def sha(path):
    with open(path, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def run(tests):
    if os.path.exists(TMP):
        os.remove(TMP)
    cmd = ['timeout', '500', 'npx', 'vitest', 'run', *tests, '--reporter=json', '--outputFile=' + TMP]
    subprocess.run(cmd, cwd=ROOT, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    try:
        with open(TMP) as f:
            r = json.load(f)
    except (OSError, ValueError):
        return None
    failed = []
    for file in r.get('testResults', []):
        name = os.path.relpath(file.get('name', ''), ROOT)
        assertions = file.get('assertionResults', [])
        if not assertions and file.get('status') == 'failed':
            failed.append(f'{name}: the file failed to load')
        for a in assertions:
            if a.get('status') == 'failed':
                failed.append(f"{os.path.basename(name)} > {a.get('fullName') or a.get('title')}")
    return r.get('numFailedTests', 0), r.get('numTotalTests', 0), failed


def read(path):
    with open(path, 'rb') as f:
        return f.read().decode('utf8')


def write(path, text):
    with open(path, 'wb') as f:
        f.write(text.encode('utf8'))

if CHECK:
    print('\n'.join(problems) or f'all {len(M)} old strings occur once')
    sys.exit(1 if problems else 0)

before = {p: sha(p) for p in files}

lines = ['# Mutations: each an exact replacement in product code (its old string occurs once), the named tests run',
         '# (npx vitest run <tests> --reporter=json), then the replacement reversed on the file as it then stands, and its sha256 checked.',
         f'# Working tree at {head} + this change; {time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())}.',
         '# sha256 before:'] + [f'#   {before[p]}  {p}' for p in files] + ['']
red = 0
for mid, desc, path, old, new, tests in M:
    start = sha(path)
    text = read(path)
    if text.count(old) != 1:
        lines.append(f'{mid}  NOT APPLIED: the old string occurs {text.count(old)} times in {path}')
        continue
    write(path, text.replace(old, new, 1))
    try:
        result = run(tests)
    finally:
        now = read(path)
        if new and now.count(new) == 1 and now.count(old) == 0:
            write(path, now.replace(new, old, 1))
        else:
            # An empty or repeated replacement cannot be found to reverse: put back the bytes read before it.
            write(path, text)
    restored = sha(path) == start
    if result is None:
        lines.append(f'{mid}  {desc}\n     {path}\n     NO RESULT (the run produced no report); restored: {restored}')
        continue
    nf, nt, failed = result
    red += 1 if nf > 0 or any('failed to load' in x for x in failed) else 0
    lines.append(f'{mid}  {desc}\n     {path}\n     tests: {", ".join(os.path.basename(t) for t in tests)}\n'
                 f'     RED {nf} of {nt}; restored and sha256 equal: {restored}')
    for x in failed[:6]:
        lines.append(f'       - {x}')
    if len(failed) > 6:
        lines.append(f'       - … and {len(failed) - 6} more')
    lines.append('')
    print(mid, nf, nt, restored, flush=True)

after = {p: sha(p) for p in files}
lines += [f'# {red} of {len(M)} mutations turned a test red.', '# sha256 after (each equal to before):']
lines += [f'#   {after[p]}  {p}  {"equal" if after[p] == before[p] else "DIFFERENT"}' for p in files]
with open(OUT, 'w') as f:
    f.write('\n'.join(lines) + '\n')
print('done', red, len(M))
