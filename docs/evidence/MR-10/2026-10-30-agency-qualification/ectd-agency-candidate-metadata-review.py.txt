"""Read existing public synthetic qualification ZIPs; never publish or mutate them."""
from pathlib import Path
import datetime
import hashlib
import json
import re
import zipfile

root = Path('/workspace/ClinicalSageAI-2-replit/docs/evidence/MR-10/2026-10-30-agency-qualification')
normalizations = [
    (rb'<xmp:(ModifyDate|CreateDate|MetadataDate)>[^<]+', rb'<xmp:\1>RUN_DATE'),
    (rb"xmpMM:DocumentID='uuid:[^']+'", b"xmpMM:DocumentID='uuid:RUN_ID'"),
    (rb'/(CreationDate|ModDate)\([^)]*\)', rb'/\1(RUN_DATE)'),
    (rb'/ID\s*\[\s*<[^>]+>\s*<[^>]+>\s*\]', b'/ID [<RUN_ID><RUN_ID>]'),
]

def normalized_pdf(data):
    for pattern, replacement in normalizations:
        data = re.sub(pattern, replacement, data)
    return data

def digest(data):
    return hashlib.sha256(data).hexdigest()

def pdf_metadata(data):
    patterns = {
        'producer': rb'/Producer\(([^)]*)\)',
        'xmpModifyDate': rb'<xmp:ModifyDate>([^<]+)',
        'xmpCreateDate': rb'<xmp:CreateDate>([^<]+)',
        'xmpMetadataDate': rb'<xmp:MetadataDate>([^<]+)',
        'xmpDocumentId': rb"xmpMM:DocumentID='([^']+)'",
        'pdfCreationDate': rb'/CreationDate\(([^)]*)\)',
        'pdfModDate': rb'/ModDate\(([^)]*)\)',
        'trailerId': rb'/ID\s*\[\s*<([^>]+)>\s*<([^>]+)>\s*\]',
    }
    values = {}
    for key, pattern in patterns.items():
        found = re.search(pattern, data)
        values[key] = [value.decode('latin1') for value in found.groups()] if found else []
    return values

sequences = []
pdf_pairs = []
for sequence in ['0000', '0001']:
    before_path = root / 'unconfigured-candidate' / f'synthetic-{sequence}.zip'
    after_path = root / 'final-unconfigured-candidate' / f'synthetic-{sequence}.zip'
    with zipfile.ZipFile(before_path) as before, zipfile.ZipFile(after_path) as after:
        row = {
            'sequence': sequence,
            'beforeZip': str(before_path.relative_to(root)),
            'afterZip': str(after_path.relative_to(root)),
            'beforeZipSha256': digest(before_path.read_bytes()),
            'afterZipSha256': digest(after_path.read_bytes()),
            'membershipIdentical': set(before.namelist()) == set(after.namelist()),
        }
        for name in before.namelist():
            if not name.endswith('.pdf'):
                continue
            old, new = before.read(name), after.read(name)
            pdf_pairs.append({
                'sequence': sequence,
                'file': name,
                'beforeBytes': len(old),
                'afterBytes': len(new),
                'beforeSha256': digest(old),
                'afterSha256': digest(new),
                'rawBytesIdentical': old == new,
                'byteIdenticalAfterOnlyListedMetadataNormalization': normalized_pdf(old) == normalized_pdf(new),
                'beforeMetadata': pdf_metadata(old),
                'afterMetadata': pdf_metadata(new),
            })
        xml_comparisons = []
        for name in ['index.xml', 'm1/us/us-regional.xml']:
            old, new = before.read(name), after.read(name)
            xml_comparisons.append({
                'file': name,
                'rawBytesIdentical': old == new,
                'byteIdenticalAfterOnlyChecksumAttributeNormalization':
                    re.sub(rb'checksum="[a-fA-F0-9]{32}"', b'checksum="DIGEST"', old) ==
                    re.sub(rb'checksum="[a-fA-F0-9]{32}"', b'checksum="DIGEST"', new),
            })
        row['xmlComparisons'] = xml_comparisons
        row['utilIndexMd5IdenticalAfterOnlyLeadingDigestNormalization'] = (
            re.sub(rb'(?m)^[a-fA-F0-9]{32}', b'DIGEST', before.read('util/index-md5.txt')) ==
            re.sub(rb'(?m)^[a-fA-F0-9]{32}', b'DIGEST', after.read('util/index-md5.txt'))
        )
        row['stfComparisons'] = [
            {'file': name, 'rawBytesIdentical': before.read(name) == after.read(name),
             'beforeSha256': digest(before.read(name)), 'afterSha256': digest(after.read(name))}
            for name in before.namelist() if re.search(r'/stf-[^/]+\.xml$', name)
        ]
        row['changedArchiveTimestamps'] = [
            {'file': name, 'beforeDateTime': list(before.getinfo(name).date_time),
             'afterDateTime': list(after.getinfo(name).date_time)}
            for name in before.namelist()
            if before.getinfo(name).date_time != after.getinfo(name).date_time
        ]
        sequences.append(row)
