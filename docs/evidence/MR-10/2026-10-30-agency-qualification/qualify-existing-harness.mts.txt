import { promises as fs } from 'node:fs';
import { createHash } from 'node:crypto';
import { qualifyV3 } from '/workspace/ClinicalSageAI-2-replit/server/services/ectd/qualification/qualify.ts';
const work = '/tmp/ectd-agency-qualification-candidate';
await fs.mkdir(work, { recursive: true });
const report = await qualifyV3('fda', work);
await fs.writeFile(`${work}/raw-qualification.json`, JSON.stringify(report, null, 2) + '\n');
const digest = async (path: string) => createHash('sha256').update(await fs.readFile(path)).digest('hex');
const originals = report.package.path;
const next = originals.replace('0000', '0001');
const assessment = {
  kind: 'actual unconfigured local qualification, synthetic golden packages',
  agencyQualification: 'blocked',
  rawAvailableChecksPassed: report.passed,
  limitations: ['Official DTD/XSL corpus absent; no schema acceptance claimed', 'Licensed external validator/profile not configured', 'Raw specification reference declares older criteria4.5, not current4.7 coverage', 'No human witness, FDA transmission or agency ACK'],
  original: { path: originals, sha256: await digest(originals) },
  amendment: { path: next, sha256: await digest(next) },
  reports: report.validators.map(({ name, ran, passed, errorCount }) => ({ name, ran, passed, errorCount })),
  notes: report.notes,
};
await fs.writeFile(`${work}/assessment.json`, JSON.stringify(assessment, null, 2) + '\n');
console.log(JSON.stringify({ agencyQualification: assessment.agencyQualification, rawAvailableChecksPassed: report.passed, original: assessment.original.sha256, amendment: assessment.amendment.sha256 }));
