// @vitest-environment jsdom
// Verifier's scratch test (never in the repo tree). Renders the FINAL card for every combination of
// dispatch status x basis x filed over two sequences, with and without documents on file and with and
// without a through-sequence, and runs every rendered header + lead through the repo's own section-9
// word list (clientNamesSlice1.shared.ts notInClientCopy) and the plan's "no stored value on screen" rule.
import React from 'react';
import { afterEach, describe, expect, it, vi } from 'vitest';
import { cleanup, render, screen } from '@testing-library/react';
import { writeFileSync } from 'node:fs';
const apiRequest = vi.hoisted(() => vi.fn());
vi.mock('@/lib/queryClient', async (importOriginal) => ({ ...(await importOriginal<typeof import('@/lib/queryClient')>()), apiRequest }));
import { notInClientCopy } from '/home/user/ClinicalSageAI-2-replit/client/src/concept2cure/v2/__tests__/clientNamesSlice1.shared';
import { EctdFiledState } from '/home/user/ClinicalSageAI-2-replit/client/src/concept2cure/v2/surfaces/EctdFiledState';

const ok = (body: unknown) => ({ ok: true, status: 200, json: async () => body }) as Response;
const STATUSES = ['sent', 'acknowledged', 'rejected', 'pending', 'transmitting', null] as const;
const BASES = ['transmit', 'compiles', null, undefined] as const;
const LEAF = { ctdSection: '3.2.S.1', fileName: 'general.pdf', title: 'General', href: 'm3/general.pdf', sequenceNumber: '0000', operation: 'new', md5: 'a'.repeat(32), history: [] };
type Seq = { sequenceNumber: string; dispatchStatus: string | null; filed: boolean; basis?: string | null };
const one = (n: string): Seq[] => {
  const out: Seq[] = [];
  for (const d of STATUSES) for (const b of BASES) for (const filed of [true, false]) {
    // The server never says filed for a status other than sent/acknowledged, nor a basis when not filed.
    if (filed && d !== 'sent' && d !== 'acknowledged') continue;
    if (!filed && b != null) continue;
    out.push({ sequenceNumber: n, dispatchStatus: d, filed, ...(b === undefined ? {} : { basis: b }) });
  }
  return out;
};
afterEach(() => { cleanup(); apiRequest.mockReset(); });

describe('every header and lead the final card can render', () => {
  it('passes the client-copy words and shows no stored status or basis value', async () => {
    const views: Array<{ sequences: Seq[]; leaves: unknown[]; throughSequence: string | null }> = [];
    for (const a of one('0000')) for (const b of one('0001')) {
      const filedAny = a.filed || b.filed;
      for (const withLeaves of filedAny ? [true, false] : [false]) {
        views.push({ sequences: [a, b], leaves: withLeaves ? [LEAF] : [], throughSequence: filedAny ? (b.filed ? '0001' : '0000') : null });
      }
    }
    views.push({ sequences: [], leaves: [], throughSequence: null });
    const texts = new Set<string>();
    for (const v of views) {
      apiRequest.mockResolvedValue(ok({ projectIdent: 'p', submissionId: 1, ...v }));
      render(<EctdFiledState identPath="p" seqQuery="" isScopeCurrent={() => true} />);
      await screen.findByText((_, el) => el?.tagName === 'P' && /on file|known/.test(el.textContent ?? ''), undefined, { timeout: 5000 });
      texts.add((screen.getByTestId('ectd-filed-state').textContent ?? '').replace(/\s+/g, ' ').trim());
      cleanup();
    }
    const lines = [...texts];
    if (process.env.COPY_PROBE_FAIL) lines.push('The filing engine read 0000 (sent).');
    writeFileSync(process.env.COPY_OUT ?? '/dev/null', `${views.length} views rendered, ${lines.length} distinct texts\n${lines.join('\n')}\n`);
    const words = lines.flatMap((l) => notInClientCopy(l).map((w) => `${w}: ${l}`));
    const stored = lines.filter((l) => /\((?:sent|acknowledged|pending|rejected|transmitting|failed)[,)]|\b(?:transmit|compiles|null|undefined)\)/.test(l) || /basis|dispatch_status|leaf manifest/i.test(l));
    expect(words).toEqual([]);
    expect(stored).toEqual([]);
  });
});
