// For every sequence region transmitSequence routes (transmitRouteFor), does the fold's agency check
// (canonicalRegionOf on both sides) accept the transmittal its own gateway writes? A 'false' would make a
// post-change send of that region fall back to compiles (or refuse), never read from what it sent.
import { transmitRouteFor } from '/home/user/ClinicalSageAI-2-replit/server/services/submission-service/submission-service';
import { canonicalRegionOf } from '/home/user/ClinicalSageAI-2-replit/shared/regulatory/region-identity';
const regions = ['fda', 'jp', 'pmda', 'eu', 'ema', 'ca', 'uk', 'cn', 'au', 'ch', 'br', 'in', 'kr', 'sg'];
let bad = 0;
for (const r of regions) {
  for (const clientType of ['biotech', 'mdx']) {
    const route = transmitRouteFor(r, clientType, process.env.APP_TYPE ?? 'MAA');
    if (!route.ok) { console.log(`${r} (${clientType}): not routed (${route.reason.slice(0, 60)}…)`); continue; }
    const seq = canonicalRegionOf(r); const gw = canonicalRegionOf(route.gwRegion);
    const same = seq !== undefined && seq === gw;
    if (!same) bad++;
    console.log(`${r} (${clientType}) -> gateway ${route.gwRegion}/${route.gwName}: canonical ${String(seq)} vs ${String(gw)} => ${same ? 'read' : 'NOT READ'}`);
  }
}
console.log(`mismatches: ${bad}`);
process.exit(0);
