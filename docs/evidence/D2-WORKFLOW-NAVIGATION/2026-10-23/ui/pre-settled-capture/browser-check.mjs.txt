const send = (route, body, status = 200) => route.fulfill({ status, contentType: 'application/json', body: JSON.stringify(body) });
await page.route('**/api/**', async route => {
  const request = route.request(), path = new URL(request.url()).pathname, method = request.method();
  requests.push({ method, path, ...(method === 'PUT' ? { body: request.postDataJSON() } : {}) });
  if (method === 'GET' && path === '/api/module-subscriptions/catalog') {
    if (mode === 'error') return send(route, { error: 'Catalog temporarily unavailable' }, 503);
    if (mode === 'malformed') return send(route, { modules: 'wrong shape' });
    return send(route, { modules: mode === 'empty' ? [] : modules, launchScope: { enforced: true } });
  }
  if (method === 'GET' && path === '/api/module-subscriptions/license') return send(route, licence);
  if (method === 'PUT') {
    assert.equal(path, '/api/module-subscriptions/review/toggle');
    assert.deepEqual(request.postDataJSON(), { enabled: true, reason: 'Enable synthetic reviewer workflow access' });
    return send(route, { moduleId: 'review', enabled: true, auditTrail: { persisted: true } });
  }
  throw new Error(`Unexpected synthetic request ${method} ${path}`);
});

}
const search = () => page.getByRole('searchbox', { name: 'Find a workflow or feature', exact: true });

  const searchColors = await search().evaluate(node => {
    const style = getComputedStyle(node);
    return { color: style.color, backgroundColor: style.backgroundColor };
  });
  const luminance = color => {
    const rgb = color.match(/^rgb\((\d+), (\d+), (\d+)\)$/);
    assert.ok(rgb, `Expected opaque computed RGB, got ${color}`);
    const channels = rgb.slice(1).map(value => Number(value) / 255).map(value => value <= 0.04045 ? value / 12.92 : ((value + 0.055) / 1.055) ** 2.4);
    return channels[0] * 0.2126 + channels[1] * 0.7152 + channels[2] * 0.0722;
  };
  const levels = [luminance(searchColors.color), luminance(searchColors.backgroundColor)].sort((a, b) => a - b);
  const contrast = (levels[1] + 0.05) / (levels[0] + 0.05);
  assert.ok(contrast >= 4.5, `Dark search text contrast below 4.5:1: ${contrast}`);
  await capture('desktop-dark-search');
  checks.push({ check: 'Dark-theme feature search retains readable text against its actual computed field background', passed: true, searchColors, contrast });
  await page.locator('.c2c-v2').evaluate(node => { node.classList.remove('dark'); node.setAttribute('data-theme', 'light'); });

  await search().fill('Future diagnostics');
