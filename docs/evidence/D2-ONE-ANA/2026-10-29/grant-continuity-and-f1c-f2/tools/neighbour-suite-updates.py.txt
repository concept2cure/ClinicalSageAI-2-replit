"""Item 2: update the six neighbouring suites the folds turned red. Each edit asserts its anchor once."""
import pathlib
R = pathlib.Path('/home/user/ClinicalSageAI-2-replit')

def edit(rel, pairs):
    p = R / rel
    s = p.read_text()
    for old, new, *count in pairs:
        n = count[0] if count else 1
        assert s.count(old) == n, (rel, old[:80], s.count(old))
        s = s.replace(old, new)
    p.write_text(s)
    print('edited', rel)
