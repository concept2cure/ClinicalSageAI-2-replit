def md5(p): return hashlib.md5(open(p, 'rb').read()).hexdigest()

def run(ids):
  out = []
  for (name, f, old, new, tests) in M:
    mid = name.split()[0]
    if ids and mid not in ids: continue
    p = os.path.join(R, f)
    src = open(p).read()
    olds, news = (old, new) if isinstance(old, list) else ([old], [new])
    if any(src.count(o) != 1 for o in olds):
      out.append(f'{name}: SKIP (anchor counts {[src.count(o) for o in olds]})'); print(out[-1], flush=True); continue
    before = md5(p)
    mutated = src
    for o, n in zip(olds, news): mutated = mutated.replace(o, n)
    open(p, 'w').write(mutated)
    try:
      cmd = tests[1:] if tests and tests[0] == 'CMD' else ['npx', 'vitest', 'run', *tests]
      r = subprocess.run(cmd, cwd=R, capture_output=True, text=True, timeout=600)
      tail = [l for l in r.stdout.splitlines() if 'Tests ' in l] or [f'exit {r.returncode}: ' + ((r.stderr or r.stdout).strip().splitlines() or [''])[-1][:160]]
      verdict = 'KILLED' if r.returncode != 0 else 'SURVIVED'
    finally:
      open(p, 'w').write(src)
    assert md5(p) == before, f'restore failed for {f}'
    out.append(f'{name}: {verdict} ({tail[-1].strip() if tail else "no summary"}) [{f} restored, md5 {before}]')
    print(out[-1], flush=True)
  return out

if __name__ == '__main__':
  res = run(set(sys.argv[1:]))
  killed = sum('KILLED' in l for l in res)
  print(f'{killed}/{len(res)} killed')
