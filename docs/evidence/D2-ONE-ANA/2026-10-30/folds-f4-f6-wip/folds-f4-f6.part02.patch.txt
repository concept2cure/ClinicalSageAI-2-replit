+#!/usr/bin/env python3
+"""Apply each mutation alone, run its tests (vitest files) or gate commands, restore the file byte for byte.
+
+Usage: mutate.py SPEC.json OUT.txt
+SPEC: [{"id", "what", "file", "old", "new", "test": str|[str], "cmd": str|[str] (optional)}]. `old` must occur exactly once.
+A mutation is RED when every test and every command exits non-zero.
+"""
+import hashlib, json, re, subprocess, sys, pathlib
+
+REPO = pathlib.Path('/home/user/ClinicalSageAI-2-replit')
+spec = json.load(open(sys.argv[1]))
+out = open(sys.argv[2], 'w')
+only = set(sys.argv[3:])
+
+def sha(b): return hashlib.sha256(b).hexdigest()
+
+for m in spec:
+    if only and m['id'] not in only: continue
+    f = REPO / m['file']
+    orig = f.read_bytes()
+    text = orig.decode()
+    n = text.count(m['old'])
+    if n != 1:
+        out.write(f"{m['id']}: SKIPPED, old text occurs {n} times in {m['file']}\n\n"); out.flush(); continue
+    f.write_bytes(text.replace(m['old'], m['new']).encode())
+    results = []
+    try:
+        tests = m.get('test', [])
+        tests = tests if isinstance(tests, list) else [tests]
+        for t in tests:
+            p = subprocess.run(['npx', 'vitest', 'run', '--maxWorkers=2', t], cwd=REPO, capture_output=True, text=True, timeout=900)
+            o = p.stdout + p.stderr
+            summ = re.findall(r'^\s+Tests\s+(.*)$', o, re.M)
+            fails = re.findall(r'^ +× (.*?)(?: \d+ms)?$', o, re.M)
+            results.append(('test', t, p.returncode, summ[-1] if summ else 'no summary', fails))
+        cmds = m.get('cmd', [])
+        cmds = cmds if isinstance(cmds, list) else [cmds]
+        for c in cmds:
+            p = subprocess.run(c, shell=True, cwd=REPO, capture_output=True, text=True, timeout=900)
+            o = (p.stdout + p.stderr).strip().splitlines()
+            results.append(('cmd', c, p.returncode, ' | '.join(o[-2:])[:300], []))
+    finally:
+        f.write_bytes(orig)
+    restored = sha(f.read_bytes()) == sha(orig)
+    out.write(f"{m['id']}: {m['what']}\n  file: {m['file']}\n")
+    for kind, t, rc, summ, fails in results:
+        out.write(f"  {kind}: {t}\n  result: {'RED' if rc != 0 else 'GREEN (survived)'} ({summ})\n")
+        for x in fails[:6]:
+            out.write(f"    x {x.split(' > ', 1)[-1]}\n")
+    out.write(f"  restored byte for byte: {'yes' if restored else 'NO'} (sha256 {sha(orig)[:16]})\n\n")
+    out.flush()
+    if not restored:
+        sys.exit(f"restore failed for {m['file']}")
