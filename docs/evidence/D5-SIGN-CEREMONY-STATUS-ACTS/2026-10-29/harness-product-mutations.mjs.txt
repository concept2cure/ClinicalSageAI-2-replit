// the gate scans. No file in the repository is written. Each is run through the
// current gate (with the baseline's recorded pairs) and the pre-change gate.
// Usage: node product-mutations.mjs <repo root> <pre-change gate path>

function newVerdict(sources, ignoreRecord = false) {
  const derived = gate.repoSignedPairs(sources);
  const recorded = ignoreRecord ? [] : baseline.signedPairs?.pairs ?? [];
  const scan = gate.scanRepo({ sources, signedPairs: new Set([...derived.keys(), ...recorded]) });
  const r = gate.evaluate(scan, baseline);
  // Without a record (P1b) there is no record check either: the gate as it would be with derivation alone.
  const { unrecorded } = ignoreRecord ? { unrecorded: [] } : acts.evaluateSignedPairs(derived, recorded);
  const lines = [
    ...r.failures.flatMap((f) => f.sites.map((s) => `FAIL ${f.file}:${s.line} ${s.kind}${s.act ? ` ${s.act}` : ''}${f.allowed ? ` (baseline allows ${f.allowed}, now ${f.count})` : ''}`)),
    ...r.shrinkable.map((x) => `FAIL baseline allows more than exists: ${x.file} ${x.count} now, baseline ${x.allowed}`),
    ...unrecorded.map((u) => `FAIL signed pair not recorded: ${u.pair} (signed in ${u.file})`),
  ];
  return { red: lines.length > 0, lines };
}

function oldVerdict(file, src) {
  const r = old.evaluate({ [file]: old.scanSource(src) }, oldBaseline);
  const lines = r.failures.flatMap((f) => f.sites.map((s) => `FAIL ${f.file}:${s.line} ${s.kind}`));
  return { red: lines.length > 0, lines };
}

const base = newVerdict(real);
console.log(`unmutated tree, current gate: ${base.red ? 'RED' : 'green'}${base.lines.length ? `\n  ${base.lines.join('\n  ')}` : ''}`);
let wrong = 0;
for (const m of mutations) {
  const src = m.create ? m.replace : real.get(m.file);
  if (!m.create && src.split(m.find).length !== 2) throw new Error(`${m.id}: anchor not found exactly once in ${m.file}`);
  const mutated = m.create ? src : src.replace(m.find, m.replace);
  const sources = new Map(real);
  sources.set(m.file, mutated);
  const now = newVerdict(sources, m.ignoreRecord);
  const before = oldVerdict(m.file, mutated);
  const got = now.red ? 'red' : 'green';
  if (got !== m.expect) wrong += 1;
  console.log(`\n${m.id}  ${m.desc}\n  file: ${m.file}${m.create ? ' (new, in memory)' : ''}`);
  console.log(`  current gate:    ${got.toUpperCase()} (expected ${m.expect})${now.lines.length ? `\n    ${now.lines.join('\n    ')}` : ''}`);
  console.log(`  pre-change gate: ${before.red ? 'RED' : 'green'}${before.lines.length ? `\n    ${before.lines.join('\n    ')}` : ''}`);
}
console.log(`\n${mutations.length - wrong} of ${mutations.length} product mutations as expected on the current gate. No repository file was written.`);
process.exit(wrong ? 1 : 0);
