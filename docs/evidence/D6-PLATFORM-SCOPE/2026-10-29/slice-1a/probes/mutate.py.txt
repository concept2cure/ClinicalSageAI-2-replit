#!/usr/bin/env python3
"""Apply one mutation (exact, unique replacement), run its test, restore by reverse replacement, verify sha256."""

def sha(p): return hashlib.sha256(open(p,'rb').read()).hexdigest()
only = sys.argv[1:]
results = []
for mid, rel, old, new, test in M:
    if only and mid not in only: continue
    p = os.path.join(REPO, rel)
    s = open(p).read()
    assert s.count(old) == 1, (mid, s.count(old))
    before = sha(p)
    mutated = s.replace(old, new)
    open(p, 'w').write(mutated)
    try:
        log = os.path.join(OUT, f'{mid}.txt')
        with open(log, 'w') as fh:
            rc = subprocess.run(['timeout', '600', 'npx', 'vitest', 'run', test], cwd=REPO, stdout=fh, stderr=subprocess.STDOUT).returncode
    finally:
        cur = open(p).read()
        if cur == mutated:
            open(p, 'w').write(s)  # nobody else wrote the file meanwhile: the original bytes
        else:  # another lane edited it meanwhile: undo only this mutation
            assert cur.count(new) == 1, (mid, 'file changed meanwhile and the mutated text is not unique; restore by hand')
            open(p, 'w').write(cur.replace(new, old))
    after = sha(p)
    txt = open(log).read()
    m = re.search(r'Tests\s+(.*)', txt)
    results.append({'mutation': mid, 'file': rel, 'test': test, 'vitest_exit': rc, 'tests': m.group(1).strip() if m else 'n/a',
                     'sha_before': before, 'sha_after': after, 'restored': before == after})
    print(json.dumps(results[-1]), flush=True)
json.dump(results, open(os.path.join(OUT, 'results.json' if not only else f'results-{"-".join(only)}.json'), 'w'), indent=1)
