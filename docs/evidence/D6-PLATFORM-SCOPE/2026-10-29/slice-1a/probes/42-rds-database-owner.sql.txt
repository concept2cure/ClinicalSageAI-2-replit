-- P-72 slice 1a, added to slice 0's RDS shape (41): on RDS the master user owns the application database
-- (Terraform's db_name is created by and for the master), so it is a member of pg_database_owner there and
-- may grant on schema public. Slice 0's copy left the database owned by postgres; no file it replayed needed this.
\set ON_ERROR_STOP on
SELECT format('ALTER DATABASE %I OWNER TO c2c_rds_master', current_database()) \gexec
SELECT current_database() AS db, pg_get_userbyid(datdba) AS owner FROM pg_database WHERE datname = current_database();
SELECT r.rolname, m.rolname AS member, a.admin_option, a.inherit_option, a.set_option
  FROM pg_auth_members a JOIN pg_roles r ON r.oid = a.roleid JOIN pg_roles m ON m.oid = a.member
 WHERE m.rolname = 'c2c_rds_master' ORDER BY 1;
SELECT rolname, rolsuper, rolcreaterole, rolbypassrls, rolinherit FROM pg_roles WHERE rolname = 'c2c_rds_master';
