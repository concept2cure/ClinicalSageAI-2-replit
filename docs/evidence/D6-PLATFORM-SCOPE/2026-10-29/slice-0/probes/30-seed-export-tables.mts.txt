/**
 * P-72 slice 0: seed one row per export-discovered table for each of the two tenants (p72-a, p72-b), as the owner,
 * with the repo's catalog seeder (tests/db/tenant-purge-row-seeder.ts). Patched here (probe only) so a uuid tenant
 * key gets the organization's uuid and a text tenant key gets its id, as the policies expect. Failures are recorded.
 */
const REPO = '/home/user/ClinicalSageAI-2-replit';
const pg = (await import(`${REPO}/node_modules/pg/lib/index.js`)).default;
const { TenantRowSeeder, qualified } = await import(`${REPO}/tests/db/tenant-purge-row-seeder.ts`);
const { discoverTenantTables } = await import(`${REPO}/server/services/tenant-export/tenant-full-export.service.ts`);
const { COMPANY_DOCUMENT_EXPORT_TABLES } = await import(`${REPO}/server/services/tenant-export/company-document-envelope.ts`);

const owner = new pg.Pool({ connectionString: process.env.DATABASE_OWNER_URL, max: 2 });
const orgs = (await owner.query(`SELECT o.id, o.uuid::text AS uuid, o.slug, u.id AS user_id FROM organizations o
  JOIN organization_users ou ON ou.organization_id = o.id JOIN users u ON u.id = ou.user_id
  WHERE o.slug IN ('p72-a','p72-b') ORDER BY o.slug`)).rows;
const discovered = await discoverTenantTables(owner);
const targets = [...new Set([...discovered.map((d: any) => `${d.schema}.${d.table}`), ...COMPANY_DOCUMENT_EXPORT_TABLES.map((t: string) => `public.${t}`), 'public.c2c_template_specs'])].sort();
console.log(`discovered ${discovered.length} tables; seeding ${targets.length} targets for ${orgs.map((o: any) => o.slug).join(', ')}`);

// Probe-only patch: the seeder fills an integer tenant column with the id; give a uuid one the uuid and a text one the id.
const TENANT = ['organization_id', 'org_id', 'tenant_id'];
const proto = (TenantRowSeeder as any).prototype;
const origColumnValues = proto.columnValues;
proto.columnValues = async function (row: Record<string, unknown>, shape: any) {
  for (const c of shape.columns) {
    if (row[c.name] !== undefined || c.generated || !TENANT.includes(c.name)) continue;
    if (c.type === 'uuid') row[c.name] = this.__uuid;
    else if (/^(text|character varying)/.test(c.type)) row[c.name] = String(this.organizationId);
  }
  return origColumnValues.call(this, row, shape);
};
const origKeyValues = proto.keyValues;
proto.keyValues = async function (t: string, shape: any, depth: number, optionalKeys: boolean) {
  const row = await origKeyValues.call(this, t, shape, depth, optionalKeys);
  for (const fk of shape.fks) if (fk.parent === 'public.organizations') fk.cols.forEach((c: string, i: number) => {
    if (row[c] !== undefined && fk.pcols[i] === 'uuid') row[c] = this.__uuid;
  });
  return row;
};

const reference = new Map();
const exercised = new Set(targets);
const failures: string[] = [];
let ok = 0;
for (const o of orgs) {
  const seeder = new TenantRowSeeder({ db: owner, organizationId: o.id, userId: o.user_id, exercised, reference, tag: `p72${o.slug.slice(-1)}` });
  (seeder as any).__uuid = o.uuid;
  for (const t of targets) {
    try { await seeder.seed(t); ok += 1; } catch (e: any) { failures.push(`${o.slug} | ${t} | ${String(e.message).replace(/\s+/g, ' ').slice(0, 160)}`); }
  }
  console.log(`${o.slug}: rows written ${seeder.rows.length}; seeded without optional keys ${seeder.withoutOptionalKeys.length}`);
}
console.log(`targets seeded ${ok} of ${targets.length * orgs.length}; failures ${failures.length}`);
for (const f of failures) console.log(`FAIL | ${f}`);
await owner.end();
