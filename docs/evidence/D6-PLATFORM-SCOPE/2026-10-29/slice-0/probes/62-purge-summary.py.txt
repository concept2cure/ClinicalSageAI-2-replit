"""P-72 slice 0, item 4: summarise purge runs. Per run: doors (counts or refusal) and the DELETE loop, then the tenant's
rows that existed before and survive after; tenant A (the control) must be untouched."""
import json, sys
for path in sys.argv[1:]:
    r = json.load(open(path))
    print(f"\n## {path.rsplit('/', 1)[-1]}  (scope {r['scope']}; purged organization {r['B']}, control {r['A']})")
    doors = [s for s in r['steps'] if s['label'].startswith('door')]
    dels = [s for s in r['steps'] if s['label'].startswith('delete')]
    print('### doors')
    for s in doors: print(f"  {s['label']}: {('ERR ' + s['err']) if 'err' in s else s.get('rows')}")
    matched = [s for s in dels if s.get('n')]; errs = [s for s in dels if 'err' in s]
    print(f"### DELETE loop: {len(dels)} statements; matched rows in {len(matched)}; deleted {sum(s.get('n') or 0 for s in dels)} rows; errors {len(errs)}")
    for s in errs[:15]: print(f"  ERR {s['label']}: {s['err'][:140]}")
    b0 = {t: v[0] for t, v in r['before'].items()}; b1 = {t: v[0] for t, v in r['after'].items()}
    a0 = {t: v[1] for t, v in r['before'].items()}; a1 = {t: v[1] for t, v in r['after'].items()}
    had = [t for t in b0 if isinstance(b0[t], int) and b0[t] > 0]
    surv = [t for t in had if isinstance(b1[t], int) and b1[t] > 0]
    ctl = [t for t in a0 if a0[t] != a1[t]]
    print(f"### organization {r['B']}: tables holding its rows before {len(had)}; still holding rows after {len(surv)}; rows before {sum(b0[t] for t in had)}, after {sum(b1[t] for t in surv)}")
    print('  survivors: ' + ', '.join(f"{t}={b1[t]}" for t in surv))
    print(f"### control organization {r['A']}: tables whose row count changed: {len(ctl)} {ctl[:10]}")
