"""P-72 slice 0: join the table-visibility runs. Rows that exist (owner) vs what the system scope sees as app_service,
with the P arm and with it removed, on the superuser-owned copy and on the RDS-shaped copy; plus p72-a's own scope."""
import sys, os
d = sys.argv[1]
def load(name):
    m = {}
    for line in open(os.path.join(d, name)):
        if ' | ' not in line: continue
        t, v = line.rstrip('\n').rsplit(' | ', 1)
        m[t] = v
    return m
own = load('vis-owner.txt'); cols = ['sys-seed', 'sys-noarm', 'sys-rds_arm', 'sys-rds_noarm', 'a-seed', 'a-rds_noarm']
runs = {c: load(f'vis-{c}.txt') for c in cols}
def num(v):
    try: return int(v)
    except: return None
rows = []
for t in sorted(own):
    rows.append([t, own[t]] + [runs[c].get(t, '-') for c in cols])
print('# table | rows (owner) | SYS arm | SYS arm removed | SYS arm (RDS owner) | SYS arm removed (RDS owner) | A-own arm | A-own arm removed (RDS owner)')
dep = [r for r in rows if num(r[2]) is not None and num(r[3]) is not None and num(r[2]) > num(r[3])]
dep_rds = [r for r in rows if num(r[4]) is not None and num(r[5]) is not None and num(r[4]) > num(r[5])]
blind = [r for r in rows if num(r[1]) and num(r[2]) is not None and num(r[2]) < num(r[1])]
owner_shape = [r for r in rows if r[2] != r[4] or r[3] != r[5]]
a_changes = [r for r in rows if r[6] != r[7]]
print(f'# RLS-enabled tables: {len(rows)}; with rows: {sum(1 for r in rows if num(r[1]))}')
print(f'# tables the system scope sees MORE of with P than without (superuser owner): {len(dep)}; rows lost without P: {sum(num(r[2]) - num(r[3]) for r in dep)}')
print(f'# the same on the RDS-shaped owner: {len(dep_rds)}; rows lost without P: {sum(num(r[4]) - num(r[5]) for r in dep_rds)}')
print(f'# tables whose system-scope count (with P) is already BELOW the rows that exist: {len(blind)}')
print(f'# tables where the owner shape changes the system-scope count: {len(owner_shape)}')
print(f"# tables where p72-a's own scope differs between arm (superuser) and arm removed (RDS owner): {len(a_changes)}")
print('\n## system scope still sees without P (superuser owner): table | owner | SYS arm | SYS no-arm')
for r in rows:
    if num(r[3]) and num(r[3]) > 0: print(' | '.join(r[:4]))
print('\n## system scope already blind WITH P (sees fewer than exist): table | owner | SYS arm | SYS arm RDS')
for r in blind: print(' | '.join([r[0], r[1], r[2], r[4]]))
print('\n## owner shape changes the system-scope count: table | owner | SYS arm | SYS no-arm | RDS arm | RDS no-arm')
for r in owner_shape: print(' | '.join(r[:6]))
print("\n## p72-a's own scope, arm (superuser) vs arm removed (RDS owner), where they differ: table | owner | A arm | A no-arm RDS")
for r in a_changes: print(' | '.join([r[0], r[1], r[6], r[7]]))
print('\n## every table')
for r in rows: print(' | '.join(r))
