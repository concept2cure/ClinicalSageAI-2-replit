  "set_config('app.current_user_role', $3, true), set_config('app.current_account_id', $4, true), " +

async function runAll(pool: any, scope: string[] | null, pick: (o: any) => any) {
  const c = await pool.connect();
  const out = new Map<string, number | string>();
  try {
    await c.query('BEGIN ISOLATION LEVEL REPEATABLE READ');
    await c.query("SET LOCAL TIME ZONE 'UTC'");
    if (scope) await c.query(SET_LOCAL, scope);
    for (const r of reads) {
      await c.query('SAVEPOINT s');
      try { const { rows } = await c.query(r.sql, [r.param(pick)]); out.set(r.label, rows.length); await c.query('RELEASE SAVEPOINT s'); }
      catch (e: any) { out.set(r.label, `ERR ${String(e.message).slice(0, 60)}`); await c.query('ROLLBACK TO SAVEPOINT s'); }
    }
    await c.query('ROLLBACK');
  } finally { c.release(); }
  return out;
}
