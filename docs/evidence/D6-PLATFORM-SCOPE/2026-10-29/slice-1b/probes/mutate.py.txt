only = sys.argv[1:] or None
results = []
for name, path, old, new, test in M:
    tag = name.split()[0]
    if only and tag not in only: continue
    src = open(f'{R}/{path}').read()
    if src.count(old) != 1:
        results.append({'mutation': name, 'error': f'old text occurs {src.count(old)} times'}); continue
    mf = f'{LC}/r4/mut/{tag}.{os.path.basename(path)}'
    open(mf, 'w').write(src.replace(old, new))
    log = f'{LC}/r4/mut/{tag}.log'
    p = subprocess.run([f'{LC}/swapn.sh', log, path, mf, '--', *V, test], capture_output=True, text=True)
    restored = 'RESTORE MISMATCH' not in p.stdout and p.returncode == 0
    tail = open(log).read()
    failed = [l.strip() for l in tail.splitlines() if l.strip().startswith('×')]
    summary = next((l.strip() for l in tail.splitlines() if l.strip().startswith('Tests ')), '')
    killed = 'command exit 0' not in p.stdout
    results.append({'mutation': name, 'file': path, 'test': test, 'killed': killed, 'restored': restored, 'tests': summary, 'red': failed[:3]})
    print(json.dumps(results[-1]), flush=True)
json.dump(results, open(f'{LC}/r4/mut/results.json', 'w'), indent=1)
