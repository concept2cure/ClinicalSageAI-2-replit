// Walk 2, flow 4b: invitations and the organisations list in Account. walk2.admin and walk2.member (accounts of the
// seeded organisation) are invited into the walk's own organisation; one accepts and switches, one declines.
const c = require('./lib.cjs');
const links = JSON.parse(c.fs.readFileSync(c.SCRATCH + '/walk2/.setup-links.json', 'utf8'));
const dlg = (page) => page.locator('[role=dialog]').first();
const memberships = () => c.psql(`select u.email, ou.organization_id, ou.role from organization_users ou join users u on u.id=ou.user_id where u.email like 'walk2.%' order by 1,2`);
const invitations = () => c.psql(`select string_agg(column_name, ',' order by ordinal_position) from information_schema.columns where table_name='organization_invitations'`) + '\n' + c.psql(`select id, organization_id, email, role, status from organization_invitations order by id`);
async function openAccount(page, name) {
  await page.goto(`${c.BASE}/concept2cure`, { waitUntil: 'networkidle' }); await c.sleep(1000);
  await page.locator('button', { hasText: name }).first().click(); await c.sleep(500);
  await page.locator('[role=menu]').getByText('Account', { exact: true }).first().click(); await c.sleep(1800);
}
(async () => {
  const browser = await c.chromium.launch();
  for (const email of ['walk2.admin@example.test', 'walk2.member@example.test']) {
    await c.step(`activate ${email} (seeded organisation) from its setup link`, async (s) => {
      if (c.psql(`select must_change_password from users where email='${email}'`) === 'f') { s.notes.push('already active'); return; }
      const ctx = await browser.newContext(); const p = await ctx.newPage();
      await p.goto(links[email], { waitUntil: 'networkidle' }); await c.sleep(1000);
      await p.fill('#reset-password', process.env.WALK_PASSWORD); await p.fill('#reset-password-confirm', process.env.WALK_PASSWORD);
      await p.getByRole('button', { name: 'Update password' }).click(); await c.sleep(2500);
      s.notes.push('after: ' + (await c.textOf(p, 'body')).split('Română').pop().slice(0, 160).replace(/\n+/g, ' | '));
      await ctx.close();
    });
  }
  const owner = (await c.pageFor(browser, 'w2owner', 'walk2.owner@example.test')).page;
  for (const [email, name, role] of [['walk2.admin@example.test', 'Wanda Walker', 'reviewer'], ['walk2.member@example.test', 'Miles Member', 'viewer']]) {
    await c.step(`own org invites ${email} (an existing account elsewhere) as ${role}`, async (s) => {
      await owner.goto(`${c.BASE}/concept2cure/admin-console`, { waitUntil: 'networkidle' }); await c.sleep(1200);
      await owner.getByRole('button', { name: 'Invite member' }).click(); await c.sleep(600);
      const d = dlg(owner);
      await d.getByLabel('Full name').fill(name); await d.getByLabel('Email').fill(email); await d.getByLabel('Role').selectOption(role);
      await d.getByRole('button', { name: 'Send invite' }).click(); await c.sleep(2500);
      const t = await c.mainText(owner, 20000);
      s.notes.push('after: ' + t.slice(Math.max(0, t.indexOf('Invitation handover')), Math.max(0, t.indexOf('Invitation handover')) + 500).replace(/\n+/g, ' | '));
      await owner.setViewportSize({ width: 1440, height: 1500 }); await c.sleep(300);
      if (role === 'reviewer') await c.shot(owner, 'w2-44-invite-existing-account-result.png');
      await owner.setViewportSize({ width: 1440, height: 900 });
      s.notes.push('memberships:\n' + memberships());
      s.notes.push('invitations:\n' + invitations());
    });
  }
  const admin = (await c.pageFor(browser, 'w2admin', 'walk2.admin@example.test')).page;
  await c.step('the invited person: Account shows the invitation; accept it', async (s) => {
    s.notes.push('signed in at ' + admin.url() + ' in: ' + (await admin.locator('button', { hasText: 'Wanda Walker' }).first().innerText().catch(() => '')).replace(/\n+/g, ' | '));
    await openAccount(admin, 'Wanda Walker');
    const d = dlg(admin);
    const t = await d.innerText();
    s.notes.push('Invitations: ' + t.slice(t.indexOf('Invitations'), t.indexOf('Reporting role')).replace(/\n+/g, ' | '));
    await c.shot(admin, 'w2-44-account-invitation-pending.png');
    await d.getByRole('button', { name: /^Accept/ }).first().click(); await c.sleep(1200);
    const confirm = admin.locator('[role=alertdialog], [role=dialog]').last();
    s.notes.push('after Accept click: ' + (await confirm.innerText()).slice(0, 500).replace(/\n+/g, ' | '));
    const yes = admin.getByRole('button', { name: /^(Accept invitation|Confirm|Accept)$/ }).last();
    if (await yes.count()) { await yes.click().catch(() => {}); await c.sleep(2500); }
    const t2 = await dlg(admin).innerText();
    s.notes.push('Invitations and switch after: ' + t2.slice(t2.indexOf('Invitations'), t2.indexOf('Reporting role')).replace(/\n+/g, ' | '));
    await c.shot(admin, 'w2-44-account-invitation-accepted.png');
    s.notes.push('memberships:\n' + memberships());
    s.notes.push('audit (org 7): ' + c.psql(`select action, actor_id, left(new_values::text, 200) from audit_logs where tenant_id=7 order by occurred_at desc limit 2`));
  });
  await c.step('switch organisation to the one just joined, and back', async (s) => {
    const d = dlg(admin);
    const sel = d.getByLabel('Organisation to open');
    s.notes.push('options: ' + JSON.stringify(await sel.locator('option').allInnerTexts()));
    await sel.selectOption({ label: 'Walk Two Biologics' }).catch(async () => { const o = await sel.locator('option').allInnerTexts(); await sel.selectOption({ label: o.find((x) => /Walk Two/.test(x)) }); });
    await d.getByRole('button', { name: 'Switch organisation' }).click(); await c.sleep(4000);
    s.notes.push('after switch: url ' + admin.url() + ' | account button: ' + (await admin.locator('aside button, nav button').filter({ hasText: 'Wanda Walker' }).first().innerText().catch(() => '')).replace(/\n+/g, ' | '));
    const me = await c.api(admin, 'GET', '/api/users/me'); s.notes.push('me: ' + JSON.stringify(me.body).slice(0, 400));
    const adm = await c.api(admin, 'GET', '/api/mdx/admin'); s.notes.push('as a reviewer of the new organisation, GET /api/mdx/admin: ' + adm.status);
    await c.shot(admin, 'w2-44-after-switch.png');
    s.notes.push('audit of the switch: ' + c.psql(`select tenant_id, action, left(new_values::text, 200) from audit_logs where user_id=3 order by occurred_at desc limit 2`));
    await openAccount(admin, 'Wanda Walker');
    const d2 = dlg(admin); const sel2 = d2.getByLabel('Organisation to open');
    s.notes.push('options now: ' + JSON.stringify(await sel2.locator('option').allInnerTexts()));
    const o = await sel2.locator('option').allInnerTexts(); await sel2.selectOption({ label: o.find((x) => /Concept2Cure Therapeutics/.test(x)) });
    await d2.getByRole('button', { name: 'Switch organisation' }).click(); await c.sleep(4000);
    s.notes.push('back: ' + (await admin.locator('aside button, nav button').filter({ hasText: 'Wanda Walker' }).first().innerText().catch(() => '')).replace(/\n+/g, ' | '));
  });
  const member = (await c.pageFor(browser, 'w2seedmember', 'walk2.member@example.test')).page;
  await c.step('the other invited person declines', async (s) => {
    await openAccount(member, 'Miles Member');
    const d = dlg(member); const t = await d.innerText();
    s.notes.push('Invitations: ' + t.slice(t.indexOf('Invitations'), t.indexOf('Reporting role')).replace(/\n+/g, ' | '));
    await d.getByRole('button', { name: /^Decline/ }).first().click(); await c.sleep(1200);
    const confirm = member.locator('[role=alertdialog], [role=dialog]').last();
    s.notes.push('after Decline click: ' + (await confirm.innerText()).slice(0, 400).replace(/\n+/g, ' | '));
    const yes = member.getByRole('button', { name: /^(Decline invitation|Confirm|Decline)$/ }).last();
    if (await yes.count()) { await yes.click().catch(() => {}); await c.sleep(2500); }
    const t2 = await dlg(member).innerText();
    s.notes.push('after: ' + t2.slice(t2.indexOf('Invitations'), t2.indexOf('Reporting role')).replace(/\n+/g, ' | '));
    s.notes.push('invitations:\n' + invitations());
    s.notes.push('memberships:\n' + memberships());
  });
  c.report(); c.save('w2-16-invitations-orgs');
  await browser.close();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });
