// Walk 2, flow 4e: "Switch organisation" from Account on a fresh sign-in; which token does the page send to /session?
const c = require('./lib.cjs');
const dlg = (page) => page.locator('[role=dialog]').first();
const org = (auth) => { try { return JSON.parse(Buffer.from(String(auth).replace('Bearer ', '').split('.')[1], 'base64url').toString()).organizationId; } catch { return '?'; } };
(async () => {
  const browser = await c.chromium.launch();
  for (const keep of [false, true]) {
    await c.step(`switch organisation in the page (${keep ? '"Keep me signed in" ticked' : 'default sign-in'})`, async (s) => {
      const ctx = await browser.newContext({ viewport: c.VIEWPORTS.desktop }); const page = await ctx.newPage();
      const seen = [];
      page.on('request', (r) => { if (/\/auth\/(session|enterprise\/select-organization)/.test(r.url())) seen.push({ t: Date.now(), m: r.method(), u: r.url().replace(c.BASE, ''), tokenOrg: org(r.headers().authorization) }); });
      page.on('response', (r) => { if (/\/auth\/(session|enterprise\/select-organization)/.test(r.url())) { const x = seen.filter((e) => e.u === r.url().replace(c.BASE, '') && e.status === undefined).shift(); if (x) x.status = r.status(); } });
      await page.goto(c.BASE + '/concept2cure/login', { waitUntil: 'networkidle' });
      await page.fill('#login-email', 'walk2.admin@example.test'); await page.fill('#login-password', process.env.WALK_PASSWORD);
      if (keep) await page.getByLabel('Keep me signed in').check();
      const since = Date.now() - 2000;
      await page.getByRole('button', { name: 'Sign in', exact: true }).click(); await page.getByLabel('Digit 1').waitFor({ timeout: 20000 });
      const code = await c.otpFor('walk2.admin@example.test', since);
      for (let i = 0; i < 6; i += 1) await page.getByLabel(`Digit ${i + 1}`).fill(code[i]); await c.sleep(400);
      const v = page.getByRole('button', { name: 'Verify', exact: true }); if (/\/login/.test(page.url()) && await v.isEnabled().catch(() => false)) await v.click().catch(() => {});
      await page.waitForURL((u) => !u.pathname.endsWith('/login'), { timeout: 20000 }); await page.waitForLoadState('networkidle'); await c.sleep(1500);
      s.notes.push('token kept in: ' + await page.evaluate(() => (localStorage.getItem('trialsage_access_token') ? 'localStorage' : sessionStorage.getItem('trialsage_access_token') ? 'sessionStorage' : 'neither')));
      s.notes.push('signed in to: ' + (await page.locator('button', { hasText: 'Wanda Walker' }).first().innerText()).replace(/\n+/g, ' | '));
      await page.locator('button', { hasText: 'Wanda Walker' }).first().click(); await c.sleep(500);
      await page.locator('[role=menu]').getByText('Account', { exact: true }).first().click(); await c.sleep(1800);
      const d = dlg(page); const sel = d.getByLabel('Organisation to open');
      const o = await sel.locator('option').allInnerTexts(); await sel.selectOption({ label: o.find((x) => !/current/.test(x)) });
      await d.getByRole('button', { name: 'Switch organisation' }).click(); await c.sleep(800);
      seen.length = 0;
      await d.getByRole('button', { name: 'Confirm switch' }).click(); await c.sleep(5000);
      s.notes.push('requests after "Confirm switch": ' + JSON.stringify(seen.map(({ m, u, tokenOrg, status }) => `${m} ${u} token-org=${tokenOrg} -> ${status}`)));
      s.notes.push('lands on: ' + page.url());
      s.notes.push('screen: ' + (await c.textOf(page, 'body')).split('Română').pop().slice(0, 330).replace(/\n+/g, ' | '));
      await c.shot(page, `w2-45-switch-org-${keep ? 'kept' : 'default'}-lands.png`);
      await ctx.close();
    });
  }
  c.report(); c.save('w2-16d-switch-ui');
  await browser.close();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });
