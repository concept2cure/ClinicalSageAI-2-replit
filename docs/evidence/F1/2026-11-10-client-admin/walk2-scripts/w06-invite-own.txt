// Walk 2: in the walk's own organization, invite a member, a viewer and a manager, and activate each from the setup link.
const c = require('./lib.cjs');
const PEOPLE = [
  { who: 'w2member', name: 'Miles Member', email: 'walk2.member2@example.test', role: 'member' },
  { who: 'w2viewer', name: 'Vera Viewer', email: 'walk2.viewer@example.test', role: 'viewer' },
  { who: 'w2manager', name: 'Mona Manager', email: 'walk2.manager@example.test', role: 'manager' },
];
(async () => {
  const browser = await c.chromium.launch();
  const { page } = await c.pageFor(browser, 'w2owner', 'walk2.owner@example.test');
  for (const p of PEOPLE) {
    let link = null;
    await c.step(`own org: invite ${p.email} as ${p.role}`, async (s) => {
      await page.goto(`${c.BASE}/concept2cure/admin-console`, { waitUntil: 'networkidle' }); await c.sleep(1500);
      await page.getByRole('button', { name: 'Invite member' }).click(); await c.sleep(600);
      const dlg = page.locator('[role=dialog]').first(); const scope = (await dlg.count()) ? dlg : page;
      await scope.getByLabel('Full name').fill(p.name); await scope.getByLabel('Email').fill(p.email);
      await scope.getByLabel('Role').selectOption(p.role);
      await scope.getByRole('button', { name: 'Send invite' }).click(); await c.sleep(2500);
      const t = await c.mainText(page, 20000);
      s.notes.push('after: ' + t.slice(Math.max(0, t.indexOf('Invitation handover')), Math.max(0, t.indexOf('Invitation handover')) + 500));
      // "Copy setup link" puts it on the clipboard; read it from the page's own state instead
      const html = await page.locator('main').first().innerHTML();
      const inputs = await page.locator('main input').evaluateAll(els => els.map(e => e.value));
      link = inputs.find(v => /password-reset\?/.test(v)) || (html.match(/https?:\/\/[^"'<\s]+password-reset\?[^"'<\s]+/) || [])[0]?.replace(/&amp;/g, '&') || null;
      if (!link) {
        await page.context().grantPermissions(['clipboard-read', 'clipboard-write']);
        await page.getByRole('button', { name: new RegExp('Copy setup link for ' + p.email.replace(/\./g, '\\.')) }).click(); await c.sleep(400);
        link = await page.evaluate(() => navigator.clipboard.readText()).catch(() => null);
      }
      s.notes.push('setup link captured: ' + Boolean(link && /password-reset/.test(link)));
      const store = c.SCRATCH + '/walk2/.setup-links-own.json'; const all = c.fs.existsSync(store) ? JSON.parse(c.fs.readFileSync(store, 'utf8')) : {}; all[p.email] = link; c.fs.writeFileSync(store, JSON.stringify(all), { mode: 0o600 });
    });
    await c.step(`activate ${p.email} from the setup link`, async (s) => {
      const ctx = await browser.newContext({ viewport: c.VIEWPORTS.desktop }); const p2 = await ctx.newPage();
      await p2.goto(link.replace('http://127.0.0.1:5231', c.BASE), { waitUntil: 'networkidle' }); await c.sleep(1200);
      s.notes.push('setup screen: ' + (await c.textOf(p2, 'body')).split('Română')[1]?.slice(0, 500));
      if (p.role === 'member') await p2.screenshot({ path: `${c.SHOTS}/w2-00e-setup-password-screen.png` });
      await p2.fill('#reset-password', process.env.WALK_PASSWORD); await p2.fill('#reset-password-confirm', process.env.WALK_PASSWORD);
      await p2.getByRole('button', { name: 'Update password' }).click(); await c.sleep(2500);
      s.notes.push('after: ' + (await c.textOf(p2, 'body')).split('Română')[1]?.slice(0, 400));
      await ctx.close();
    });
  }
  c.report(); c.save('w2-06-invite-own');
  await browser.close();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });
