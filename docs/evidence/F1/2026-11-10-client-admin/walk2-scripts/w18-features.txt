// Walk 2, flow 5a: "Subscription & features" in the walk's own organisation (free plan): what it says, and the figures.
const c = require('./lib.cjs');
(async () => {
  const browser = await c.chromium.launch();
  const { page } = await c.pageFor(browser, 'w2owner', 'walk2.owner@example.test');
  await c.step('Subscription & features: as found', async (s) => {
    await page.goto(`${c.BASE}/concept2cure/features`, { waitUntil: 'networkidle' }); await c.sleep(2500);
    await c.shot(page, 'w2-50-features-1440.png');
    await c.tallShot(page, 'w2-50-features-1440-tall.png', 3400);
    const lic = await c.api(page, 'GET', '/api/module-subscriptions/license'); s.notes.push('GET license: ' + JSON.stringify(lic.body).slice(0, 900));
    const cat = await c.api(page, 'GET', '/api/module-subscriptions/catalog'); s.notes.push('GET catalog: ' + lic.status + ' ' + JSON.stringify(cat.body).slice(0, 1200));
    s.notes.push('db: members ' + c.psql('select count(*) from organization_users where organization_id=7') + ' | pending invitations ' + c.psql(`select count(*) from organization_invitations where organization_id=7 and status='pending'`) + ' | org row ' + c.psql('select tier, max_users, max_projects, max_storage from organizations where id=7') + ' | projects ' + c.psql('select count(*) from projects where organization_id=7'));
    s.notes.push('db module_subscriptions: ' + c.psql(`select string_agg(column_name, ',' order by ordinal_position) from information_schema.columns where table_name='module_subscriptions'`));
    s.notes.push(c.psql(`select module_id, status, coalesce(tier,''), is_active from module_subscriptions where organization_id=7 order by 1`).slice(0, 1500));
    s.notes.push('db licenses: ' + c.psql(`select string_agg(column_name, ',' order by ordinal_position) from information_schema.columns where table_name='licenses'`) + '\n' + c.psql('select * from licenses where organization_id=7').slice(0, 600));
  });
  await c.step('Subscription & features: open every section', async (s) => {
    for (const b of await page.locator('main button:visible').filter({ hasText: /^(Feature access|Features and steps)/ }).all()) { await b.click().catch(() => {}); await c.sleep(250); }
    const add = page.locator('main button:visible').filter({ hasText: /^Additional capabilities/ }).first(); if (await add.count()) { await add.click().catch(() => {}); await c.sleep(400); }
    s.notes.push(await c.mainText(page, 16000));
    await c.tallShot(page, 'w2-50-features-open-1440-tall.png', 5200);
  });
  await c.step('Subscription & features: Admin controls', async (s) => {
    await page.getByRole('button', { name: 'Admin controls' }).click(); await c.sleep(1500);
    const t = await c.mainText(page, 24000);
    s.notes.push(t.slice(0, 7000));
    s.notes.push('controls: ' + JSON.stringify(await page.locator('main button:visible, main [role=switch]:visible, main input[type=checkbox]:visible').evaluateAll(els => els.map(e => `${e.tagName.toLowerCase()}${e.getAttribute('role') ? '[' + e.getAttribute('role') + ']' : ''}:${(e.getAttribute('aria-label') || e.textContent || '').trim().replace(/\s+/g, ' ').slice(0, 70)}${e.disabled ? ' [disabled]' : ''}${e.getAttribute('aria-checked') ? ' checked=' + e.getAttribute('aria-checked') : ''}`)).slice(0, 80)));
    await c.shot(page, 'w2-50-features-admin-controls-1440.png');
    await c.tallShot(page, 'w2-50-features-admin-controls-1440-tall.png', 5200);
  });
  c.report(); c.save('w2-18-features');
  await browser.close();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });
