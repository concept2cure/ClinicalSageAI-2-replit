// RFC 6238 time-based code (SHA-1, 30 s, 6 digits) from a base32 key, as an authenticator app computes it.
const crypto = require('crypto');
function base32(s) { const A = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567'; let bits = ''; for (const ch of s.replace(/=+$/, '').replace(/\s+/g, '').toUpperCase()) { const v = A.indexOf(ch); if (v < 0) throw new Error('not base32'); bits += v.toString(2).padStart(5, '0'); } const out = []; for (let i = 0; i + 8 <= bits.length; i += 8) out.push(parseInt(bits.slice(i, i + 8), 2)); return Buffer.from(out); }
function totp(secret, at = Date.now(), stepOffset = 0) { const counter = Math.floor(at / 30000) + stepOffset; const buf = Buffer.alloc(8); buf.writeBigUInt64BE(BigInt(counter)); const h = crypto.createHmac('sha1', base32(secret)).update(buf).digest(); const o = h[h.length - 1] & 0xf; const n = ((h[o] & 0x7f) << 24) | (h[o + 1] << 16) | (h[o + 2] << 8) | h[o + 3]; return String(n % 1000000).padStart(6, '0'); }
module.exports = { totp };
