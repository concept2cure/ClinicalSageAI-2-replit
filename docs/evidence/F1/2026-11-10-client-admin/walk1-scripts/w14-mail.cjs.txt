// On the second instance (5231, mail to the local sink): what the product's mails say. Sign-in code, invitation to a
// new person, and the sign-up refusal for an address that already has an account.
const c = require('../walk/common.cjs');
const mail = require('./mail.cjs');
const note = (k, v) => console.log(`## ${k}:`, typeof v === 'string' ? v : JSON.stringify(v));
const red = (s) => String(s).replace(/token=[A-Za-z0-9._%-]+/g, 'token=<redacted>').replace(/\b\d{6}\b/g, '<code>').replace(/"(token|setupUrl)":"[^"]*"?/g, '"$1":"<redacted>"');
const AVERY = 'avery.lind@northgate-bio.example.test';
(async () => {
  const browser = await c.chromium.launch();
  const ctx = await browser.newContext({ viewport: c.VIEWPORTS.desktop }); const page = await ctx.newPage();
  // sign-up with an address that already has an account
  const dup = await page.request.post(c.BASE + '/api/auth/signup', { data: { email: AVERY, password: process.env.WALK_PASSWORD, companyName: 'Second Northgate', industryMode: 'biotech', firstName: 'Avery', lastName: 'Lind' } });
  note('sign-up with an address that already has an account', `${dup.status()} ${(await dup.text()).slice(0, 300)}`);
  // sign in on this instance: the code arrives by mail
  await page.goto(c.BASE + '/concept2cure/login', { waitUntil: 'networkidle' });
  await page.fill('#login-email', AVERY); await page.fill('#login-password', process.env.WALK_PASSWORD);
  let since = Date.now() - 1500;
  await page.getByRole('button', { name: 'Sign in', exact: true }).click();
  await page.getByLabel('Digit 1').waitFor({ timeout: 20000 });
  const m = await mail.mailFor(AVERY, since, '');
  note('sign-in code mail: subject', red(mail.subjectOf(m)).slice(0, 60));
  note('sign-in code mail: text', red(mail.plainOf(m)).slice(0, 500));
  const code = mail.codeIn(m);
  for (let i = 0; i < 6; i += 1) await page.getByLabel(`Digit ${i + 1}`).fill(code[i]);
  await c.sleep(400);
  if (/\/login/.test(page.url())) { const v = page.getByRole('button', { name: 'Verify', exact: true }); if (await v.isEnabled().catch(() => false)) await v.click().catch(() => {}); }
  await page.waitForURL((u) => !u.pathname.endsWith('/login'), { timeout: 20000 }); await c.sleep(1500);
  // free a seat (Marta, invited) and invite a new person; read the mail
  note('remove Marta (invited) to free a seat', await c.api(page, 'DELETE', '/api/tenant-users/6/13', { reason: 'Invitation withdrawn to free a seat for the mail check' }).then(r => `${r.status} ${JSON.stringify(r.body).slice(0, 160)}`));
  since = Date.now() - 1500;
  const inv = await c.api(page, 'POST', '/api/tenant-users', { name: 'Lena Fischer', email: 'lena.fischer@northgate-bio.example.test', role: 'member' });
  note('invite Lena (mail configured)', red(`${inv.status} ${JSON.stringify(inv.body)}`).slice(0, 500));
  const im = await mail.mailFor('lena.fischer@northgate-bio.example.test', since, '').catch(e => null);
  if (im) { note('invitation mail: subject', mail.subjectOf(im)); note('invitation mail: from/to', `${im.from} -> ${im.to}`); note('invitation mail: text', red(mail.plainOf(im)).slice(0, 1200)); }
  else note('invitation mail', 'none arrived in the sink');
  // welcome mail sent at verification?
  note('mails so far (subjects)', mail.mails().map(x => `${x.time.slice(11, 19)} ${x.to.join(',')} | ${mail.subjectOf(x)}`));
  await browser.close();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });
