const wide = (page) => page.evaluate(() => {
  const out = [];
  for (const e of document.querySelectorAll('main *, [role=dialog] *, form *')) {
    const r = e.getBoundingClientRect(); if (!r.width || !r.height) continue;
    if (r.right > innerWidth + 2 && r.left < innerWidth) out.push(`${e.tagName.toLowerCase()}.${String(e.className).split(' ')[0]} right=${Math.round(r.right)} "${(e.innerText || '').trim().slice(0, 30)}"`);
  }
  return [...new Set(out)].slice(0, 8);
});
