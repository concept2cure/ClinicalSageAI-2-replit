// A whole sign-up through the page, typed one key at a time, on the second instance (port 5231, mail to a local sink).
// The password is read from a file named by WALK_PW_FILE and is never printed. Nothing here is a real account.
const { chromium } = require('playwright');
const fs = require('fs');
const OUT = '/home/user/ClinicalSageAI-2-replit/docs/evidence/F1/2026-11-10-client-admin/fix-signup';
const PW = fs.readFileSync(process.env.WALK_PW_FILE, 'utf8').trim();
const EMAIL = process.env.WALK_EMAIL;
(async () => {
  const browser = await chromium.launch();
  const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 }, deviceScaleFactor: 2 });
  const page = await ctx.newPage();
  let sent = null;
  let answer = null;
  page.on('request', (r) => {
    if (!r.url().includes('/api/auth/signup')) return;
    try {
      const body = JSON.parse(r.postData());
      sent = Object.fromEntries(Object.entries(body).filter(([k]) => k !== 'password'));
    } catch { /* not JSON */ }
  });
  page.on('response', (r) => { if (r.url().includes('/api/auth/signup')) answer = r.status(); });
  await page.goto('http://127.0.0.1:5231/concept2cure/signup', { waitUntil: 'networkidle' });
  const type = async (sel, text) => { await page.click(sel); await page.keyboard.type(text, { delay: 15 }); };
  await type('#firstName', 'Morgan');
  await type('#lastName', 'Hale');
  await type('#email', EMAIL);
  await type('#password', PW);
  await type('#confirmPassword', PW);
  await type('#jobTitle', 'Head of Regulatory Affairs');
  await page.getByRole('button', { name: /continue/i }).click();
  await type('#organization', 'Hale Diagnostics');
  await page.selectOption('#organizationType', 'meddevice');
  await type('#country', 'United States');
  await page.selectOption('#useCase', { index: 1 });
  await page.getByRole('button', { name: /continue/i }).click();
  await page.getByRole('button', { name: /continue/i }).click();
  await page.waitForSelector('input[type=checkbox]');
  for (let pass = 0; pass < 3; pass += 1) {
    await page.evaluate(() => {
      document.querySelectorAll('[class*="overflow-y-auto"]').forEach((b) => {
        b.scrollTop = b.scrollHeight;
        b.dispatchEvent(new Event('scroll', { bubbles: true }));
      });
    });
    await page.waitForTimeout(250);
    for (const cb of await page.$$('input[type=checkbox]')) {
      if (!(await cb.isChecked()) && !(await cb.isDisabled())) await cb.check();
    }
  }
  await page.screenshot({ path: `${OUT}/signup-1440-agreements-accepted.png`, fullPage: true });
  await page.getByRole('button', { name: /create account/i }).click();
  await page.waitForTimeout(4000);
  await page.screenshot({ path: `${OUT}/signup-1440-after-create.png` });
  const after = (await page.evaluate(() => document.body.innerText)).slice(0, 300).replace(/\n+/g, ' | ');
  console.log(JSON.stringify({ sent, answer, after }, null, 1));
  await browser.close();
})().catch((e) => { console.error('FAILED', e); process.exit(1); });
