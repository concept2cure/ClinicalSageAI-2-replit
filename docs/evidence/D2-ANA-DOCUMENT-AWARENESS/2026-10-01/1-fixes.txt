 * reaches Postgres is exactly the mutated one:
 *   SELECT id FROM regulatory_programs WHERE id = $1 AND organization_id = $2 AND deleted_at IS NULL LIMIT 1
 *   → SELECT id FROM regulatory_programs WHERE id = $1 AND deleted_at IS NULL LIMIT 1
 */

  const real = (await importOriginal()) as typeof import('/home/user/ClinicalSageAI-2-replit/server/services/c2c/program-access.ts');
  return {
    ...real,
    resolveOpenProgram: (pool: { query: (sql: string, params?: unknown[]) => Promise<{ rows: any[] }> }, ctx: any) =>
      real.resolveOpenProgram({
        query: (sql: string, params?: unknown[]) => {
          if (/FROM regulatory_programs/.test(sql) && sql.includes(' AND organization_id = $2')) {

        },
      }, ctx),
