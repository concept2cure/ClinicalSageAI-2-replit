import {chromium} from '/tmp/vault-handoff-tools/node_modules/playwright-core/index.mjs';
import {mkdir,writeFile} from 'node:fs/promises';
import assert from 'node:assert/strict';
const out='/workspace/ClinicalSageAI/docs/evidence/D2/2026-10-30-vault-final-review';
await mkdir(out,{recursive:true});
const PROGRAM='11111111-1111-4111-8111-111111111111', DOCUMENT='22222222-2222-4222-8222-222222222222';
const make=()=>({success:true,document:{id:DOCUMENT,programId:PROGRAM,documentCode:'receipt.pdf',documentTitle:'receipt',version:'1.0',extraction:'extracted'},filing:{folderId:'module-3',folderLabel:'Module 3 · Quality',placementStatus:'suggested',needsReview:false}});
const cases=['invalid-json','missing-id','wrong-project','wrong-code','valid-new','valid-unchanged','valid-repaired','explicit-unknown','gateway-error'];
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});
const results=[];
try{
for(const width of [1440,390])for(const theme of ['light','dark'])for(const kind of cases){
 const ctx=await browser.newContext({viewport:{width,height:1000},colorScheme:theme});
 const page=await ctx.newPage();page.setDefaultTimeout(12000);const errors=[];page.on('pageerror',e=>errors.push(e.message));
 let writes=0,headersMatched=true,contextMatched=true;
 await page.route('**/api/vault/ingest',async route=>{
  const req=route.request(),h=req.headers();writes++;
  headersMatched&&=Boolean(h.authorization?.startsWith('Bearer '))&&h['x-organization-id']==='7';
  contextMatched&&=req.method()==='POST'&&/multipart\/form-data; boundary=/.test(h['content-type']??'')&&(req.postData()??'').includes(PROGRAM)&&(req.postData()??'').includes('receipt.pdf');
  const body=make();let status=201,contentType='application/json';
  if(kind==='gateway-error'){await route.fulfill({status:502,contentType:'text/html',body:'<p>Bad gateway</p>'});return;}
  if(kind==='invalid-json'){await route.fulfill({status:201,contentType:'text/html',body:'<!doctype html><p>synthetic invalid JSON reply</p>'});return;}
  if(kind==='missing-id')delete body.document.id;
  if(kind==='wrong-project')body.document.programId='33333333-3333-4333-8333-333333333333';
  if(kind==='wrong-code')body.document.documentCode='other.pdf';
  if(kind==='valid-unchanged'){
   status=200;body.document.documentTitle=null;
   body.filing={folderId:'module-3',folderLabel:'Module 3 · Quality',placementStatus:'unfiled',needsReview:true};
   body.reupload={unchanged:true,changes:[],differs:[{field:'document_title',recorded:null,requested:'receipt'}]};
  }
  if(kind==='valid-repaired'){status=200;body.reupload={unchanged:false,changes:[{field:'storage_version_id',from:null,to:'stored-copy'}],differs:[]};}
  if(kind==='explicit-unknown'){await route.fulfill({status:503,contentType,body:JSON.stringify({error:{code:'OUTCOME_UNKNOWN',message:'The Vault admission outcome could not be confirmed. Review the current Vault records before trying again.'}})});return;}
  await route.fulfill({status,contentType,body:JSON.stringify(body)});
 });
 try{
  await page.goto('http://127.0.0.1:5197');
  await page.evaluate(dark=>{document.documentElement.classList.toggle('dark',dark);document.querySelector('.c2c-v2').classList.toggle('dark',dark);},theme==='dark');
  await page.getByTestId('vault-stage-upload').click();
  await page.waitForFunction(()=>document.querySelector('[data-testid="outcome"]')?.getAttribute('data-value')!=='null');
  const outcome=JSON.parse(await page.getByTestId('outcome').getAttribute('data-value'));
  const row=await page.getByTestId('vault-stage-row').innerText(),note=await page.getByTestId('batch-note').innerText();
  const unknown=!kind.startsWith('valid-');
  if(unknown){assert.deepEqual(outcome.succeeded,[]);assert.deepEqual(outcome.recorded,[]);assert.equal(outcome.failed[0].unknown,true);assert.match(row,/not known whether/i);assert.match(note,/not known whether/i);assert(!/Not filed|Nothing partial was recorded/.test(note));}
  else{assert.deepEqual(outcome.recorded,[DOCUMENT]);assert.equal(outcome.succeeded.length,1);assert.deepEqual(outcome.failed,[]);}
  if(kind==='valid-new'){assert.match(row,/suggested, awaiting confirmation/);assert.match(note,/Auto-filed.*suggested until confirmed/);}
  if(kind==='valid-unchanged'){assert.match(row,/already in the Vault/i);assert(!/Auto-filed|Module 3|\(as/.test(note));assert.equal(outcome.alreadyRecorded[0].title,null);}
  if(kind==='valid-repaired'){assert.match(row,/record was updated/);assert.match(note,/record was updated/);assert(!/nothing changed|Nothing was changed/.test(row+note));assert.equal(outcome.alreadyRecorded[0].updated,true);}
  assert.equal(writes,1);assert.equal(headersMatched,true);assert.equal(contextMatched,true);assert.deepEqual(errors,[]);
  const overflow=await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth);assert.equal(overflow,false);
  if(['gateway-error','valid-repaired'].includes(kind))await page.screenshot({path:`${out}/receipt-${kind}-${width}-${theme}.png`,fullPage:true});
  results.push({width,theme,kind,unknown,writes,headersMatched,contextMatched,overflow,pageErrors:errors});
 }catch(error){console.error(JSON.stringify({width,theme,kind,writes,headersMatched,contextMatched,errors,body:await page.locator('body').innerText()},null,2));throw error;}
 finally{await ctx.close();}
}
await writeFile(`${out}/browser-check.json`,JSON.stringify({seams:'Actual useVaultUpload, sendIngest XMLHttpRequest/JSON.parse, receipt parser, staging UI, shared header utility and canonical CSS. Only HTTP replies and JWT/file are synthetic; no hosted authenticated journey is claimed. Raw HTML201 exercises actual JSON.parse rejection.',flows:results.length,results},null,2)+'\n');
console.log(JSON.stringify({flows:results.length,results},null,2));
}finally{await browser.close();}
