import React from 'react';
import { createRoot } from 'react-dom/client';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { useLocation } from 'wouter';
import { AuthProvider, useAuth } from './src/services/portal/authService';
import { TaskTray } from './src/concept2cure/v2/TaskTray';
import { TaskBoard } from './src/concept2cure/v2/surfaces/TaskBoard';
import { locationForSurface } from './src/concept2cure/v2/routing';
import '../design-system/colors_and_type.css';
import './src/index.css';
import './src/concept2cure/v2/styles/app-v2.css';
import './src/concept2cure/v2/styles/project-home-v2.css';
function App() {
  const [, setLocation] = useLocation(); const { user } = useAuth();
  const onNav = (id: string) => setLocation(locationForSurface(id));
  return <div className="c2c-v2 shell"><main className="main"><header><TaskTray onNav={onNav} /></header><div className="page">{user && <TaskBoard {...({onNav,onAsk:()=>{},surface:{id:'tasks',label:'Tasks'},segment:'biopharma'} as any)} />}</div></main></div>;
}
createRoot(document.getElementById('root')!).render(<QueryClientProvider client={new QueryClient()}><AuthProvider><App /></AuthProvider></QueryClientProvider>);
