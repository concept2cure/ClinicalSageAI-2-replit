#!/usr/bin/env bash
source /workspace/.cloud-setup/activate.sh
cd /workspace/ClinicalSageAI-2-replit || exit 99
python3 /workspace/.cloud-setup/rbm-navigation/push-preflight.py || exit 98
python3 /workspace/.cloud-setup/rbm-navigation/capture.py push-start.json
c2c_rbm_push_head=$(git rev-parse HEAD)
git -c core.hooksPath=.husky push origin concept2cure-v2 > /workspace/.cloud-setup/rbm-navigation/code-push.txt 2>&1
c2c_rbm_push_exit=$?
python3 /workspace/.cloud-setup/rbm-navigation/capture.py push-finish.json
python3 - "$c2c_rbm_push_exit" "$c2c_rbm_push_head" <<'PYPROOF'
from pathlib import Path
import datetime,hashlib,json,sys
p=Path('/workspace/.cloud-setup/rbm-navigation')
start=json.loads((p/'push-start.json').read_text()); end=json.loads((p/'push-finish.json').read_text())
r={'processExit':int(sys.argv[1]),'finishedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'pushedCommit':sys.argv[2],'sourceCommitAtFinish':end['commit'],'unchangedInputs':start['files']==end['files'],'transcriptSha256':hashlib.sha256((p/'code-push.txt').read_bytes()).hexdigest()}
(p/'code-push-exit.json').write_text(json.dumps(r,indent=2)+'\n');print(json.dumps(r,indent=2))
PYPROOF
exit "$c2c_rbm_push_exit"
