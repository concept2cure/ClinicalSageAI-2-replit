from pathlib import Path
import datetime,hashlib,json,shlex,subprocess,sys
root=Path('/workspace/ClinicalSageAI-2-replit');p=Path('/workspace/.cloud-setup/rbm-navigation')
prefix=sys.argv[1];selection=json.loads((p/sys.argv[2]).read_text())
paths=set(selection['files'])|set(selection.get('inputs',[]))
assert all((root/f).is_file() for f in paths),[f for f in paths if not (root/f).is_file()]
def capture(suffix):
 x={'observedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'commit':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),'files':{f:hashlib.sha256((root/f).read_bytes()).hexdigest() for f in sorted(paths)}}
 (p/(prefix+'-'+suffix+'.json')).write_text(json.dumps(x,indent=2)+'\n');return x
start=capture('start')
args=['npx','vitest','run','--config','vitest.config.ts',*selection['files'],'--reporter=json','--outputFile='+str(p/(prefix+'-results.json'))]
command='source /workspace/.cloud-setup/activate.sh && NODE_OPTIONS=--max-old-space-size=4096 '+shlex.join(args)
with (p/(prefix+'-tests.txt')).open('wb') as out: run=subprocess.run(['bash','-c',command],cwd=root,stdout=out,stderr=subprocess.STDOUT)
end=capture('finish');data=json.loads((p/(prefix+'-results.json')).read_text())
x={'command':command,'exitCode':run.returncode,'passed':data['numPassedTests'],'failed':data['numFailedTests'],'skipped':data['numPendingTests'],'files':len(data['testResults']),'unchangedInputs':start['files']==end['files'],'failures':[{'file':f['name'],'cases':[a['fullName'] for a in f['assertionResults'] if a['status']=='failed']} for f in data['testResults'] if f['status']!='passed']}
(p/(prefix+'-summary.json')).write_text(json.dumps(x,indent=2)+'\n');print(json.dumps(x,indent=2));sys.exit(run.returncode or (0 if x['unchangedInputs'] else 98))
