from pathlib import Path
import datetime,hashlib,json,subprocess,sys
p=Path('/workspace/.cloud-setup/rbm-navigation');r=Path('/workspace/ClinicalSageAI-2-replit')
subprocess.run(['python3',str(p/'capture.py'),'final-ci-start.json'],check=True)
results=[]
for name in ['ci:launch-scope','ci:launch-scope-api','ci:sign-ceremony','ci:sql-interpolation']:
 f=p/('final-attachments-'+name.removeprefix('ci:')+'.txt')

 results.append({'command':'npm run --silent '+name,'exitCode':code,'transcript':f.name,'sha256':hashlib.sha256(f.read_bytes()).hexdigest()})
