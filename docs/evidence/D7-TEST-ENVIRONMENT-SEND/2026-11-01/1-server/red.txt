export const SEQUENCE_STATUSES = ['draft', 'assembling', 'validated', 'frozen', 'dispatched'] as const;
export type SequenceStatus = (typeof SEQUENCE_STATUSES)[number];


const SEQUENCE_TRANSITIONS: Record<SequenceStatus, SequenceStatus[]> = {
  draft: ['assembling'],
  assembling: ['validated', 'draft'],
  validated: ['frozen', 'assembling'],
  frozen: ['dispatched'],
  dispatched: [],
};


export function canTransitionSequence(from: string, to: string): boolean {
  const allowed = SEQUENCE_TRANSITIONS[from as SequenceStatus];
  return Array.isArray(allowed) && allowed.includes(to as SequenceStatus);


export function sequenceDispatchAuthorizationEligibility(seq: {
  status: string; region: string; dispatchStatus?: string | null;
}): { reauthorizing: boolean; eligible: boolean; refusal: string | null } {
  const reauthorizing = seq.status === 'dispatched' && canonicalRegionOf(seq.region) !== undefined;
  if (reauthorizing) {
    const eligible = seq.dispatchStatus === null || seq.dispatchStatus === 'pending' || seq.dispatchStatus === 'rejected';
    return { reauthorizing, eligible, refusal: eligible ? null :
      resendRefusal(seq.dispatchStatus) ?? 'The dispatch status is not confirmed as unsent; refresh it before renewing its authorization.' };
  const eligible = canTransitionSequence(seq.status, 'dispatched');
  return { reauthorizing, eligible, refusal: eligible ? null : `Cannot transition sequence from ${seq.status} to dispatched.` };

export async function listSubmissions(

export type TransitionedSequence = EctdSequence & { auditTrail: AuditRowOutcome };


export async function transitionSequence(
  id: number,
  toStatus: string,
  ctx: { organizationId: number; userId: number }
): Promise<TransitionedSequence> {
  const seq = await getSequence(id, ctx);
  if (GOVERNED_TRANSITIONS.has(toStatus)) {
    throw new SubmissionError(
      'GOVERNED_REQUIRED',
      `Transition to ${toStatus} is irreversible and must go through the governed e-signature flow (POST /api/c2c/actions sign), not this endpoint.`
    );
  if (!canTransitionSequence(seq.status, toStatus)) {
    throw new SubmissionError('INVALID_STATE', `Cannot transition sequence from ${seq.status} to ${toStatus}.`);


  const validation = toStatus === 'validated' ? await validateForValidatedStatus(seq, ctx) : null;
  const frozenAt = toStatus === 'frozen' ? new Date() : undefined;
  
  const [row] = await db
    .update(ectdSequences)
    .set({
      status: toStatus,
      updatedAt: new Date(),
      ...(frozenAt ? { frozenAt } : {}),



      ...(validation ? { validationStatus: 'passed' } : seq.status === 'validated' ? { validationStatus: null } : {}),
    })
    .where(
      and(
        eq(ectdSequences.id, id),
        eq(ectdSequences.organizationId, ctx.organizationId),
        eq(ectdSequences.status, seq.status),
      ),
    )
    .returning();
  if (!row) {
    throw new SubmissionError(
      'INVALID_STATE',
      `Sequence ${id} is no longer in state '${seq.status}' — it changed while this transition was being applied. Re-read the sequence and retry if the transition still applies.`,
    );





  const auditTrail = await recordAuditRow({
    organizationId: ctx.organizationId,
    userId: ctx.userId,
    action: toStatus === 'frozen' ? 'SEQUENCE_FROZEN' : 'SEQUENCE_TRANSITIONED',
    resourceType: 'ectd_sequence',
    resourceId: id,
    details: { from: seq.status, to: toStatus, ...(validation ? { validation } : {}) },
  return { ...(row as EctdSequence), auditTrail };


async function validateForValidatedStatus(
  seq: EctdSequence,
  ctx: { organizationId: number },
): Promise<{ errors: number; warnings: number; notYetApproved: number; leafCount: number }> {
  const before = await sequenceLeafManifestDigest(asQueryable(pool), seq.id, ctx.organizationId);
  const { assessSequenceDispatchReadiness } = await import('../ectd/assess-dispatch-readiness');
  const a = await assessSequenceDispatchReadiness({ sequenceId: seq.id, organizationId: ctx.organizationId });
  const errors = a.readiness.findings.filter((f) => f.severity === 'error');
  if (errors.length > 0) {
    const shown = errors
      .slice(0, 5)
      .map((f) => (f.sectionCode ? `${f.sectionCode}: ${f.message}` : f.message))
      .join(' ');
    throw new SubmissionError(
      'VALIDATION_FAILED',
      `Sequence ${seq.sequenceNumber} was not marked Validated: its validation found ${errors.length} ` +
        `${errors.length === 1 ? 'error' : 'errors'}. ${shown}${errors.length > 5 ? ` (and ${errors.length - 5} more in the Validation tab)` : ''}`,
    );
  const after = await sequenceLeafManifestDigest(asQueryable(pool), seq.id, ctx.organizationId);
  if (before !== after) {
    throw new SubmissionError(
      'INVALID_STATE',
      `Sequence ${seq.sequenceNumber}'s leaves or recorded application/applicant facts changed while it was being validated, so it was not marked Validated. Validate it again.`,
    );
  return { errors: 0, warnings: a.readiness.warnings, notYetApproved: a.readiness.releaseBlockers, leafCount: a.leafCount };

type SignaturePayloadRefusal = (payload: Record<string, unknown> | null) => string | null;


async function governedSignatureVerdict(
  signatureActionId: string,
  target: string,
  ctx: { organizationId: number; userId: number },
  step: SequenceSignatureIntent,
  payloadRefusal?: SignaturePayloadRefusal,
): Promise<
  | { refusal: string; boundDigest: null }
  | { refusal: null; boundDigest: string; signatureId: number | null; meaning: unknown; recordedPackageIdentity?: RecordedPackageIdentitySnapshot }
> {
  const refuse = (refusal: string) => ({ refusal, boundDigest: null }) as const;
  const action = await db.execute(sql`
    SELECT id, payload FROM c2c_ana_actions
    WHERE id = ${signatureActionId}
      AND org_id = ${ctx.organizationId}
      AND command = 'sign'
      AND target = ${target}
      AND state = 'executed'
      AND proposed_by = ${ctx.userId}
    LIMIT 1
  `);
  const row = ((action as { rows?: Array<{ payload?: unknown }> }).rows ?? [])[0];
  if (!row) return refuse('no executed sign action on this sequence by this actor');

  const payload = typeof row.payload === 'string' ? safeJson(row.payload) : (row.payload as Record<string, unknown> | null);
  const intent = typeof payload?.intent === 'string' ? payload.intent : null;
  if (intent !== step) {
    return refuse(`the sign action declares intent '${intent ?? 'none'}', not '${step}'; sign this step with its own meaning`);








  if (!requiresIndependence('sign', payload?.meaning)) {
    return refuse(
      `the signature declares the meaning '${String(payload?.meaning)}'; ${INTENT_LABEL[step]} is an approval or release step, ` +
        'so it must be signed with an approval, responsibility or release meaning by someone other than the ' +
        'sequence\\'s creator',
    );


  const intentMeaning = INTENT_MEANING_REFUSAL[step]?.(payload?.meaning) ?? null;
  if (intentMeaning) return refuse(intentMeaning);


  const notThisRequest = payloadRefusal?.(payload ?? null) ?? null;
  if (notThisRequest) return refuse(notThisRequest);



  const spentBy = await signatureSpentBy(signatureActionId, target, ctx.organizationId);
  if (spentBy === SIGNATURE_VOIDED_ACTION) {
    return refuse(
      `this signature was given for a ${INTENT_LABEL[step]} the server refused, so it is void; sign again for a new attempt`,
    );
  if (spentBy !== null) {
    return refuse('this sign action already authorized a governed transition; each step needs its own signature');

  
  const esig = await db.execute(sql`
    SELECT id, bound_payload_digest, binding_basis, superseded_by, is_valid, verification_status
      FROM electronic_signatures
    WHERE organization_id = ${ctx.organizationId}
      AND signed_target = ${target}
      AND (signature_manifest::jsonb ->> 'actionId') = ${signatureActionId}
    LIMIT 1
  `);
  const sig = ((esig as unknown as {
    rows?: Array<{
      id: unknown;
      bound_payload_digest: string | null;
      binding_basis: string | null;
      superseded_by: unknown;
      is_valid: unknown;
      verification_status: unknown;
    }>;
  if (!sig) return refuse('no electronic signature record is bound to this sign action');
  if (String(sig.verification_status ?? '') === VOIDED_VERIFICATION_STATUS) {
    return refuse(
      `this signature was given for a ${INTENT_LABEL[step]} the server refused, so it is void; sign again for a new attempt`,
    );
  if (isSignatureWithdrawn(sig)) {
    return refuse('the electronic signature authorizing this step has been revoked; obtain a new signature');
  if (sig.binding_basis !== BINDING_BASIS.ECTD_SEQUENCE_LEAF_MANIFEST || !sig.bound_payload_digest) {
    return refuse('the signature is not bound to this sequence\\'s leaf manifest; re-sign the sequence');
  const current = await deriveGovernedTargetBinding(
    { query: (text: string, params?: unknown[]) => pool.query(text, params) as Promise<{ rows: any[] }> },
    target,
    ctx.organizationId,
  );
  if (current.digest !== sig.bound_payload_digest) {
    return refuse('the sequence changed after it was signed (leaves or recorded application/applicant facts differ); re-sign the current content');
  const signatureId = Number(sig.id);
  return {
    refusal: null,
    boundDigest: sig.bound_payload_digest,
    signatureId: Number.isFinite(signatureId) && signatureId > 0 ? signatureId : null,
    meaning: payload?.meaning,
    ...(current.recordedPackageIdentity ? { recordedPackageIdentity: current.recordedPackageIdentity } : {}),

function safeJson(text: string): Record<string, unknown> | null {
  try { return JSON.parse(text) as Record<string, unknown>; } catch { return null; }


const SIGNATURE_VOIDED_ACTION = 'GOVERNED_SIGNATURE_VOIDED';


async function signatureSpentBy(signatureActionId: string, target: string, organizationId: number, client?: PoolClient): Promise<string | null> {
  const sequenceId = target.slice(target.indexOf(':') + 1);
  const query = sql`
    SELECT action FROM audit_logs
    WHERE tenant_id = ${organizationId}
      AND table_name = 'ectd_sequence'
      AND record_id = ${sequenceId}
      AND action IN ('SEQUENCE_FROZEN', 'SEQUENCE_DISPATCHED', 'ECTD_TRANSMITTED', ${SHADOW_REVIEW_WAIVED_ACTION},
                     ${SHADOW_FINDING_DISPOSITIONED_ACTION}, ${SIGNATURE_VOIDED_ACTION})
      AND (new_values::jsonb ->> 'signatureActionId') = ${signatureActionId}
    LIMIT 1
  `;
  const spent = client ? await drizzle(client).execute(query) : await db.execute(query);
  const row = ((spent as { rows?: Array<{ action?: unknown }> }).rows ?? [])[0];
  return row ? String(row.action ?? '') : null;


type SignatureVoidOutcome = 'voided' | 'not-voidable' | 'failed';


async function voidRefusedStepSignature(
  signatureActionId: string,
  sequenceId: number,
  ctx: { organizationId: number; userId: number },
  step: SequenceSignatureIntent,
  refusal: string,
): Promise<SignatureVoidOutcome> {
  const target = `ectd-sequence:${sequenceId}`;
  try {
    const action = await db.execute(sql`
      SELECT payload FROM c2c_ana_actions
      WHERE id = ${signatureActionId}
        AND org_id = ${ctx.organizationId}
        AND command = 'sign'
        AND target = ${target}
        AND state = 'executed'
        AND proposed_by = ${ctx.userId}
      LIMIT 1
    `);
    const row = ((action as { rows?: Array<{ payload?: unknown }> }).rows ?? [])[0];
    if (!row) return 'not-voidable';
    const payload = typeof row.payload === 'string' ? safeJson(row.payload) : (row.payload as Record<string, unknown> | null);
    if (payload?.intent !== step) return 'not-voidable';
    if ((await signatureSpentBy(signatureActionId, target, ctx.organizationId)) !== null) return 'not-voidable';
    logger.error('A refused governed step could not read the signature it was given, so it was not voided', {
      sequenceId, step, signatureActionId, organizationId: ctx.organizationId,
      err: err instanceof Error ? err.message : String(err),
    });
    return 'failed';

  let signatureId: number | null = null;
  let signatureRowFailed = false;
  try {
    signatureId = await voidSignatureForRefusedAct(asQueryable(pool), {
      orgId: ctx.organizationId, target, actionId: signatureActionId, occurredAt: new Date(),
    });
    signatureRowFailed = true;
    logger.error('A refused governed step could not void its signature row', {
      sequenceId, step, signatureActionId, organizationId: ctx.organizationId,
      err: err instanceof Error ? err.message : String(err),
    });


  if (signatureId === null && !signatureRowFailed) return 'not-voidable';

  const audit = await recordAuditRow({
    organizationId: ctx.organizationId,
    userId: ctx.userId,
    action: SIGNATURE_VOIDED_ACTION,
    resourceType: 'ectd_sequence',
    resourceId: sequenceId,
    details: { signatureActionId, signatureId, signatureRowVoided: signatureId !== null, step, refusal },
  if (signatureId !== null || audit.persisted) return 'voided';
  logger.error('A refused governed step voided its signature nowhere', { sequenceId, step, signatureActionId, organizationId: ctx.organizationId });
  return 'failed';


function voidedSignatureSentence(outcome: SignatureVoidOutcome, step: SequenceSignatureIntent): string {
  if (outcome === 'voided') return ` The signature given for this ${INTENT_LABEL[step]} is now void; sign again for a new attempt.`;
  if (outcome === 'failed') {
    return ` The signature given for this ${INTENT_LABEL[step]} could not be marked void right now; do not reuse it, sign again for a new attempt.`;
  return '';


async function voidingSignatureOnRefusal<T>(
  run: () => Promise<T>,
  sig: { signatureActionId: string; sequenceId: number; ctx: { organizationId: number; userId: number }; step: SequenceSignatureIntent },
  voidable: (err: unknown) => boolean | Promise<boolean> = () => true,
): Promise<T> {
  try {
    return await run();
    if (!(await voidable(err))) throw err;
    const refusal = err instanceof Error ? err.message : String(err);
    const outcome = await voidRefusedStepSignature(sig.signatureActionId, sig.sequenceId, sig.ctx, sig.step, refusal);
    const sentence = voidedSignatureSentence(outcome, sig.step);
    if (!sentence) throw err;
    if (err instanceof SubmissionError) throw new SubmissionError(err.code, `${err.message}${sentence}`);
    if (err instanceof Error) err.message = `${err.message}${sentence}`;
    throw err;

type LifecycleKeyLeaf = {
  section_code: string | null;
  lifecycle_op: string | null;
  document_table: string | null;
  document_id: number | null;
  document_uuid: string | null;
};


function sharedLifecycleKeys(
  mine: LifecycleKeyLeaf[],
  theirs: LifecycleKeyLeaf[],
  leafSourceKey: (table: string | null | undefined, id: number | null | undefined, uuid?: string | null) => string,
): string | null {
  const sectionOf = (l: LifecycleKeyLeaf) => String(l.section_code ?? '').trim().toLowerCase();
  const documentOf = (l: LifecycleKeyLeaf): string | null =>
    l.document_table && String(l.document_table).trim() && (l.document_id || l.document_uuid)
      ? leafSourceKey(String(l.document_table).trim(), l.document_id, l.document_uuid).toLowerCase()
      : null;
  const bindsBySection = (l: LifecycleKeyLeaf) =>
    documentOf(l) === null && String(l.lifecycle_op ?? '').trim().toLowerCase() === 'delete';
  const found: string[] = [];
  const sections = new Set(mine.map(sectionOf));
  for (const section of [...sections].sort()) {
    const a = mine.filter((l) => sectionOf(l) === section);
    const b = theirs.filter((l) => sectionOf(l) === section);
    if (b.length === 0) continue;
    const label = String(a[0].section_code ?? '').trim();
    if (a.some(bindsBySection) || b.some(bindsBySection)) {
      found.push(`${label}: a withdrawal there names no document, so it binds by section`);
      continue;
    }
    const theirDocs = new Set(b.map(documentOf).filter((k): k is string => k !== null));
    const same = [...new Set(a.map(documentOf).filter((k): k is string => k !== null && theirDocs.has(k)))].sort();
    if (same.length > 0) found.push(`${label}: document ${same.join(', ')}`);
  return found.length > 0 ? found.join('; ') : null;


async function filingOrderRefusal(
  q: SequenceQueryable,
  seq: { id: number; submissionId: number; sequenceNumber: string; status: string },
  organizationId: number,

  const res = await q.query(
    `SELECT o.id, o.sequence_number, o.status, o.dispatch_status, o.region
       FROM ectd_sequences o
      WHERE o.submission_id = $1 AND o.organization_id = $2 AND o.deleted_at IS NULL
        AND o.id <> $3 AND o.sequence_number <> $4
        AND o.status IN ('frozen', 'dispatched')
      ORDER BY o.sequence_number`,
    [seq.submissionId, organizationId, seq.id, seq.sequenceNumber],
  );
  type Other = { id: number; sequence_number: string; status: string; dispatch_status: string | null; region: string | null; shared_sections: string };
  const locked = (res.rows ?? []) as Other[];
  if (locked.length === 0) return null;
  const leafRes = await q.query(
    `SELECT sequence_id, section_code, lifecycle_op, document_table, document_id, document_uuid
       FROM submission_leaves
      WHERE organization_id = $1 AND deleted_at IS NULL AND sequence_id = ANY($2::int[])`,
    [organizationId, [seq.id, ...locked.map((r) => Number(r.id))]],
  );
  const leavesOf = new Map<number, LifecycleKeyLeaf[]>();
  for (const l of (leafRes.rows ?? []) as Array<LifecycleKeyLeaf & { sequence_id: number }>) {
    const id = Number(l.sequence_id);
    leavesOf.set(id, [...(leavesOf.get(id) ?? []), l]);
  const { leafSourceKey } = await import('../ectd/leaf-source-resolver');
  const mine = leavesOf.get(seq.id) ?? [];
  const dependent: Other[] = [];
  for (const r of locked) {
    const shared = sharedLifecycleKeys(mine, leavesOf.get(Number(r.id)) ?? [], leafSourceKey);
    if (shared) dependent.push({ ...r, shared_sections: shared });
  const isFiled = (r: Other) => isFiledDispatchStatus(r.dispatch_status);


  const lower = dependent.filter((r) => String(r.sequence_number) < seq.sequenceNumber && !isFiled(r));


  const legacyFrozenPair = (r: Other) => seq.status === 'frozen' && r.status === 'frozen' && !isFiled(r);
  const higher = dependent.filter((r) => String(r.sequence_number) > seq.sequenceNumber && !legacyFrozenPair(r));
  const state = (r: Other) =>
    `${r.status}, dispatch status '${r.dispatch_status ?? 'none'}', region '${r.region ?? 'none'}'`;
  if (lower.length > 0) {
    const first = lower[0];
    const others = lower.slice(1).map((r) => r.sequence_number);
    return (
      `Refusing to ${step}: sequence ${first.sequence_number} is not yet filed; ${step} this one after it is sent` +
      `${others.length > 0 ? ` (also not yet filed: ${others.join(', ')})` : ''}. ` +
      `It is ${state(first)}, and it files the same lifecycle key — ${first.shared_sections} — where this sequence's lifecycle ` +
      'operations are bound against what is on file — sending it changes that, so a verdict taken now would not hold at transmit. ' +
      'This is a filing-order constraint, not a defect in this sequence\\'s leaves.'
    );
  if (higher.length > 0) {
    const last = higher[higher.length - 1];
    const first = higher[0];
    const filed = isFiled(first);
    return (
      `Refusing to ${step}: sequence ${first.sequence_number} is ${filed ? 'already filed' : `${first.status} and not yet filed`} ` +
      `(${state(first)}) and files the same lifecycle key — ${first.shared_sections} — where this sequence's lifecycle operations bind. ` +
      'This sequence can no longer be filed ahead of it: it is judged only against the sequences numbered below it, ' +
      (filed
        ? `so its lifecycle operations would bind to what ${first.sequence_number} has already changed. `
        : `and ${first.sequence_number}'s lifecycle operations were judged without this one on file, so whichever is sent ` +
          'second would transmit a lifecycle judged against the wrong inventory. ') +
      `Place these leaves in a new sequence numbered after ${last.sequence_number}, where they are judged against what it files. ` +
      'This is a filing-order constraint, not a defect in this sequence\\'s leaves.'
    );
  return null;


async function assertSequencePackageable(
  id: number,
  ctx: { organizationId: number; userId: number },

): Promise<string | null> {
  const sequence = await getSequence(id, ctx);
  const binding = await deriveGovernedTargetBinding(asQueryable(pool), `ectd-sequence:${id}`, ctx.organizationId);
  const assembledAgainst = binding.digest;
  const region = canonicalRegionOf(sequence.region);
  const fda = region === 'US';
  const captured = binding.recordedPackageIdentity;
  const { assembleSequence, assembledTransmitBlockers } = await import('../ectd/assemble-from-core');
  let assembled: Awaited<ReturnType<typeof assembleSequence>>;
  try {



    if (!region) throw new Error('The sequence region is not recognized, so its recorded release identity cannot be established.');
    if (!captured) throw new Error('The signature binding has no recorded application/applicant identity.');
    const missing = packageIdentityRefusal(captured.identity, 'Nothing can be frozen or dispatched.', { fda });
    if (missing) throw missing;
    assembled = await assembleSequence({
      sequenceId: id,
      organizationId: ctx.organizationId,
      userId: ctx.userId,
      applicationId: captured.identity.applicationNumber as string,
      sponsorId: fda ? captured.identity.applicantId as string : '',
      sponsorName: captured.identity.applicantName as string,
      recordedPackageIdentity: captured,
      ...(fda ? { contacts: fdaContactsForPackageIdentity(captured.identity) } : {}),
    });



    if (isConversionCapacityError(err)) throw err;
    throw new SubmissionError(
      'DISPATCH_BLOCKED',
      `Refusing to ${step}: the sequence does not assemble into a package (${err instanceof Error ? err.message : String(err)}). ` +
        'Fix it while the leaves can still be changed.',
    );
  try {
    const gaps = assembledTransmitBlockers(assembled);
    if (gaps.length > 0) {
      throw new SubmissionError(
        'DISPATCH_BLOCKED',
        `Refusing to ${step}: ${gaps.join('; ')}. Transmit would refuse this package, and once the sequence is ` +
          `${step === 'freeze' ? 'frozen' : 'dispatched'} its leaves can no longer be changed.`,
      );
    }
    await assembled.cleanup();
  return assembledAgainst;


async function governedContentGates(
  seq: EctdSequence,
  ctx: { organizationId: number; userId: number },
  step: 'freeze' | 'dispatch',
  verdictFor: (a: DispatchReadinessAssessment) => { cleared: boolean; blockers: string[] },
): Promise<{ assessment: DispatchReadinessAssessment; assembledAgainst: string | null }> {
  const toStatus = step === 'freeze' ? 'frozen' : 'dispatched';
  const { assessSequenceDispatchReadiness } = await import('../ectd/assess-dispatch-readiness');
  const assessment = await assessSequenceDispatchReadiness({ sequenceId: seq.id, organizationId: ctx.organizationId });
  const stepGate = verdictFor(assessment);
  if (!stepGate.cleared) {
    throw new SubmissionError(
      'DISPATCH_BLOCKED',
      `Dispatch gate blocks ${toStatus}: ${stepGate.blockers.join(' ')}`
    );
  const orderRefusal = await filingOrderRefusal(asQueryable(pool), seq, ctx.organizationId, step);
  if (orderRefusal) throw new SubmissionError('DISPATCH_BLOCKED', orderRefusal);
  const assembledAgainst = await assertSequencePackageable(seq.id, ctx, step);
  return { assessment, assembledAgainst };

  const step = SHADOW_FINDING_DISPOSITION_INTENT;
  const { findingId, disposition } = opts;
  if (typeof findingId !== 'number' || !isShadowFindingDisposition(disposition)) {
    return { step, cleared: false, refusal: 'Name the finding and how it is closed (accepted or fixed); nothing was checked.' };
  const shadowFindingDisposition = await shadowFindingDisposable(asQueryable(pool), {
    sequenceId: seq.id, organizationId: ctx.organizationId, findingId,
  const refusal = shadowFindingDispositionRefusal(shadowFindingDisposition, disposition);
  return { step, cleared: refusal === null, refusal, shadowFindingDisposition };


export async function precheckGovernedStep(
  id: number,
  step: SequenceSignatureIntent,
  ctx: { organizationId: number; userId: number },
  opts: GovernedPrecheckOptions = {},
): Promise<GovernedStepPrecheck> {
  if (step === 'transmit') {
    const transmit = await sequenceTransmitReadiness(id, ctx, opts.environment, opts.applicationId);
    return { step, cleared: transmit.refusal === null, refusal: transmit.refusal, transmit };
  const seq = await getSequence(id, ctx);

  if (step === SHADOW_WAIVER_INTENT) {
    const shadowWaiver = await shadowReviewWaivable(asQueryable(pool), { sequenceId: seq.id, organizationId: ctx.organizationId });
    return { step, cleared: shadowWaiver.eligible, refusal: shadowWaiver.eligible ? null : shadowWaiver.reason, shadowWaiver };
  if (step === SHADOW_FINDING_DISPOSITION_INTENT) return precheckShadowFindingDisposition(seq, ctx, opts);
  const toStatus = step === 'freeze' ? 'frozen' : 'dispatched';
  const dispatchAuthorization = step === 'dispatch' ? sequenceDispatchAuthorizationEligibility(seq) : null;
  if (dispatchAuthorization ? !dispatchAuthorization.eligible : !canTransitionSequence(seq.status, toStatus)) {
    return { step, cleared: false, refusal: dispatchAuthorization?.refusal ?? `Cannot transition sequence from ${seq.status} to ${toStatus}.` };
  try {
    await governedContentGates(seq, ctx, step, (a) => (step === 'freeze' ? a.freezeGate : a.dispatchGateOnSigning));
    return { step, cleared: true, refusal: null };
    if (err instanceof SubmissionError && err.code === 'DISPATCH_BLOCKED') {
      return { step, cleared: false, refusal: err.message };
    }
    throw err;


function waiverReliedOn(a: Partial<Pick<DispatchReadinessAssessment, 'shadowReviewRunCount' | 'shadowReviewWaiver'>>): Record<string, unknown> | null {
  const w = a.shadowReviewWaiver;
  if (a.shadowReviewRunCount !== 0 || w?.state !== 'active') return null;
  return {
    signatureActionId: w.signatureActionId ?? null,
    signatureId: w.signatureId ?? null,
    waiverRunId: w.waiverRunId ?? null,
    failedRunId: w.failedRunId ?? null,
    failureCode: w.failureCode ?? null,


function withWaiverReliedOn(a: Parameters<typeof waiverReliedOn>[0]): { shadowReviewWaiver?: Record<string, unknown> } {
  const w = waiverReliedOn(a);
  return w ? { shadowReviewWaiver: w } : {};


async function shadowReviewRefusalUnderLock(
  q: SequenceQueryable,
  a: Partial<Pick<DispatchReadinessAssessment, 'shadowReviewRunCount' | 'shadowReviewWaiver' | 'unacknowledgedShadowCriticals'>>,
  scope: { sequenceId: number; organizationId: number },
  step: string,
): Promise<string | null> {
  const since = `Since this ${step} was checked`;
  const criticals = await criticalsStandingOpen(q, scope);
  if (criticals.count > Number(a.unacknowledgedShadowCriticals ?? 0)) {
    const why = criticals.reopened.map(reopenedCriticalSentence).join(' ');
    return `${since}, a critical Shadow Review finding blocks dispatch again. ${why}`.trim();
  if (!waiverReliedOn(a)) return null;
  const presence = evaluateShadowPresenceWithWaiver(
    await completedShadowReviewRuns(q, scope),
    await resolveShadowReviewWaiver(q, scope),
  );
  return presence.cleared ? null : `${since}, the Shadow Review waiver it relied on no longer stands. ${presence.blockers.join(' ')}`;


function assertDispatchRenewalStillEligible(
  row: { status?: string; region?: string; dispatch_status?: string | null } | undefined,
  expectedDispatchStatus: string | null | undefined,
): void {
  const current = sequenceDispatchAuthorizationEligibility({ status: row?.status ?? '',
    region: row?.region ?? '', dispatchStatus: row?.dispatch_status });
  if (!current.eligible || !current.reauthorizing || row?.dispatch_status !== expectedDispatchStatus) {
    throw new SubmissionError('INVALID_STATE',
      current.refusal ?? 'The dispatch status changed while renewal was being checked. Nothing was recorded; refresh and sign again if it remains unsent.');

async function applyGovernedSequenceTransition(
  id: number,
  toStatus: 'frozen' | 'dispatched',
  ctx: { organizationId: number; userId: number },
  signatureActionId: string
): Promise<EctdSequence> {
  const seq = await getSequence(id, ctx);
  const dispatchAuthorization = toStatus === 'dispatched' ? sequenceDispatchAuthorizationEligibility(seq) : null;
  if (dispatchAuthorization ? !dispatchAuthorization.eligible : !canTransitionSequence(seq.status, toStatus)) {
    throw new SubmissionError('INVALID_STATE', dispatchAuthorization?.refusal ?? `Cannot transition sequence from ${seq.status} to ${toStatus}.`);
  const reauthorizing = dispatchAuthorization?.reauthorizing === true;


  const target = `ectd-sequence:${id}`;
  const step: GovernedSequenceStep = toStatus === 'frozen' ? 'freeze' : 'dispatch';
  const verdict = await governedSignatureVerdict(signatureActionId, target, ctx, step);
  if (verdict.refusal !== null) {
    throw new SubmissionError(
      'GOVERNED_REQUIRED',
      `A valid e-signature is required: sign ${target} via POST /api/c2c/actions/sign with intent '${step}', then pass its actionId. Refused: ${verdict.refusal}.`
    );


  const { assessment, assembledAgainst } = await governedContentGates(
    seq,
    ctx,
    step === 'freeze' ? 'freeze' : 'dispatch',
    (a) => (toStatus === 'frozen' ? a.freezeGate : a.gate),
  );



  await applySequenceChangeWithAudit(
    {
      
      underLock: async (client) => {
        const q = asQueryable(client);
        const parent = await lockSubmissionPackageIdentity(q, ctx.organizationId, seq.submissionId, { forUpdate: true });
        const locked = await client.query(
          `SELECT id, submission_id, status, dispatch_status, region FROM ectd_sequences WHERE id = $1 AND organization_id = $2 AND deleted_at IS NULL FOR UPDATE`,
          [id, ctx.organizationId],
        );
        if (!parent || locked.rows[0]?.submission_id !== seq.submissionId ||
          parent.programId !== verdict.recordedPackageIdentity?.identity.programId) {
          throw new SubmissionError('INVALID_STATE', 'The sequence parent or the project’s tracking record changed while this release was being checked. Nothing was recorded; refresh and sign the current content.');
        }
        if (reauthorizing) {
          assertDispatchRenewalStillEligible(locked.rows[0], seq.dispatchStatus);


          if (await signatureSpentBy(signatureActionId, target, ctx.organizationId, client)) {
            throw new SubmissionError('GOVERNED_REQUIRED', 'This dispatch signature was already used or voided while renewal was being checked. Sign a fresh authorization.');
          }
        }
        await lockRecordedPackageIdentity(q, ctx.organizationId, parent.programId);
        const now = await sequenceLeafManifestDigest(q, id, ctx.organizationId);
        if (now === null || now !== assembledAgainst || now !== verdict.boundDigest) {
          throw new SubmissionError(
            'INVALID_STATE',
            `Sequence ${id}'s leaves or recorded application/applicant facts changed while this ${step} was being checked (the content binding no longer matches ` +
              'the one the package gate assembled and the signature binds). Nothing was changed; re-check the leaves and sign again.',
          );
        }
        const lockedOrderRefusal = await filingOrderRefusal(q, seq, ctx.organizationId, step);
        if (lockedOrderRefusal) throw new SubmissionError('DISPATCH_BLOCKED', lockedOrderRefusal);
        const shadowRefusal = await shadowReviewRefusalUnderLock(q, assessment, { sequenceId: id, organizationId: ctx.organizationId }, step);
        if (shadowRefusal) throw new SubmissionError('DISPATCH_BLOCKED', shadowRefusal);
      },
      
      text: toStatus === 'frozen'
        ? `UPDATE ectd_sequences SET status = $1, updated_at = NOW(), frozen_at = NOW() WHERE id = $2 AND organization_id = $3 AND deleted_at IS NULL AND status = $4`
        : reauthorizing
          ? `UPDATE ectd_sequences SET status = $1, updated_at = NOW() WHERE id = $2 AND organization_id = $3 AND deleted_at IS NULL AND status = $4 AND dispatch_status IS NOT DISTINCT FROM $5::text AND (dispatch_status IS NULL OR dispatch_status IN ('pending', 'rejected'))`
          : `UPDATE ectd_sequences SET status = $1, updated_at = NOW(), dispatch_status = 'pending' WHERE id = $2 AND organization_id = $3 AND deleted_at IS NULL AND status = $4`,
      params: [toStatus, id, ctx.organizationId, seq.status, ...(reauthorizing ? [seq.dispatchStatus] : [])],
      noRowsRefusal:
        `Sequence ${id} is no longer in state '${seq.status}' — it changed while this ${step} was being authorized. ` +
        `Re-check the sequence and sign again if the transition still applies.`,
    },
    {
      organizationId: ctx.organizationId,
      userId: ctx.userId,
      action: STEP_AUDIT_ACTION[step],
      resourceId: id,
      details: {
        from: seq.status,
        to: toStatus,
        signatureActionId,
        ...(reauthorizing ? { reauthorized: true, previousDispatchStatus: seq.dispatchStatus } : {}),
        ...(verdict.recordedPackageIdentity ? { recordedPackageIdentity: verdict.recordedPackageIdentity } : {}),
        validationErrors: assessment.validationErrors,
        unacknowledgedShadowCriticals: assessment.unacknowledgedShadowCriticals,

        ...withWaiverReliedOn(assessment),
      },
    },
  );
  logger.info('Governed sequence transition applied', { id, toStatus, organizationId: ctx.organizationId });
  return getSequence(id, ctx);

async function recheckUnderSequenceLock(
  client: PoolClient,
  p: { sequenceId: number; submissionId: number; organizationId: number; boundDigest: string; act: string; recheck: (q: SequenceQueryable) => Promise<void> },
): Promise<void> {
  const q = asQueryable(client);
  const parent = await lockSubmissionPackageIdentity(q, p.organizationId, p.submissionId);
  const locked = await client.query(
    `SELECT id, submission_id FROM ectd_sequences WHERE id = $1 AND organization_id = $2 AND deleted_at IS NULL FOR UPDATE`,
    [p.sequenceId, p.organizationId],
  );
  if (!parent || locked.rows[0]?.submission_id !== p.submissionId) {
    throw new SubmissionError('INVALID_STATE', 'The sequence parent changed while this decision was being recorded. Nothing was recorded; refresh and sign the current content.');
  await lockRecordedPackageIdentity(q, p.organizationId, parent.programId);
  await p.recheck(q);
  if ((await sequenceLeafManifestDigest(q, p.sequenceId, p.organizationId)) !== p.boundDigest) {
    throw new SubmissionError(
      'INVALID_STATE',
      `Sequence ${p.sequenceId}'s leaves or recorded application/applicant facts changed while this ${p.act} was being recorded, so the signature no longer binds them. ` +
        'Nothing was recorded; sign again over the current content.',
    );

async function applyShadowReviewWaiver(
  input: { sequenceId: number; signatureActionId: string; failedRunId: number },
  ctx: { organizationId: number; userId: number },
): Promise<ShadowReviewWaiverRecord> {
  const { signatureActionId, failedRunId } = input;
  const seq = await getSequence(input.sequenceId, ctx);
  const target = `ectd-sequence:${seq.id}`;
  const verdict = await governedSignatureVerdict(signatureActionId, target, ctx, SHADOW_WAIVER_INTENT);
  if (verdict.refusal !== null) {
    throw new SubmissionError(
      'GOVERNED_REQUIRED',
      `A valid e-signature is required: sign ${target} via POST /api/c2c/actions/sign with intent '${SHADOW_WAIVER_INTENT}' ` +
        `and the meaning responsibility or approval, then pass its actionId. Refused: ${verdict.refusal}.`,
    );
  const independence = await independentSignerRefusal(target, ctx, verdict.meaning);
  if (independence) throw independence;
  const offered = await shadowReviewWaivable(asQueryable(pool), { sequenceId: seq.id, organizationId: ctx.organizationId });
  const notOffered = shadowWaiverRunRefusal(offered, failedRunId);
  if (notOffered !== null) throw new SubmissionError('SHADOW_WAIVER_NOT_ELIGIBLE', notOffered);
  const failureCode = offered.failureCode;
  if (!isWaivableShadowReviewFailure(failureCode)) throw new SubmissionError('SHADOW_WAIVER_NOT_ELIGIBLE', offered.reason);

  const insert = shadowWaiverInsert({
    sequenceId: seq.id, organizationId: ctx.organizationId, signerId: ctx.userId, signatureActionId, failedRunId, failureCode,
  await applySequenceChangeWithAudit(
    {
      underLock: (client) =>
        recheckUnderSequenceLock(client, {
          sequenceId: seq.id,
          submissionId: seq.submissionId,
          organizationId: ctx.organizationId,
          boundDigest: verdict.boundDigest,
          act: 'waiver',
          recheck: async (q) => {
            const still = shadowWaiverRunRefusal(
              await shadowReviewWaivable(q, { sequenceId: seq.id, organizationId: ctx.organizationId }),
              failedRunId,
            );
            if (still !== null) throw new SubmissionError('SHADOW_WAIVER_NOT_ELIGIBLE', still);
          },
        }),
      text: insert.text,
      params: insert.params,
      noRowsRefusal: `Run #${failedRunId} is no longer a failed run of this sequence, so nothing was waived.`,
    },
    {
      organizationId: ctx.organizationId,
      userId: ctx.userId,
      action: SHADOW_REVIEW_WAIVED_ACTION,
      resourceId: seq.id,
      details: { signatureActionId, signatureId: verdict.signatureId, failedRunId, failureCode },
    },
  );
  logger.info('Shadow Review waived', { sequenceId: seq.id, failedRunId, failureCode, organizationId: ctx.organizationId });
  const waiver = await resolveShadowReviewWaiver(asQueryable(pool), { sequenceId: seq.id, organizationId: ctx.organizationId });
  return { sequenceId: seq.id, failedRunId, failureCode, signatureActionId, signatureId: verdict.signatureId, waiver };

async function applyShadowFindingDisposition(
  input: ShadowFindingDispositionRequest & { sequenceId: number; signatureActionId: string },
  ctx: { organizationId: number; userId: number },
): Promise<ShadowFindingDispositionRecord> {
  const { findingId, disposition, signatureActionId } = input;
  const seq = await getSequence(input.sequenceId, ctx);
  const target = `ectd-sequence:${seq.id}`;
  const verdict = await governedSignatureVerdict(signatureActionId, target, ctx, SHADOW_FINDING_DISPOSITION_INTENT, (payload) =>
    shadowFindingDispositionPayloadRefusal(payload, { findingId, disposition }),
  );
  if (verdict.refusal !== null) {
    throw new SubmissionError(
      'GOVERNED_REQUIRED',
      `A valid e-signature is required: sign ${target} via POST /api/c2c/actions/sign with intent ` +
        `'${SHADOW_FINDING_DISPOSITION_INTENT}', the payload { findingId: ${findingId}, disposition: '${disposition}' } and the ` +
        `meaning approval or responsibility, then pass its actionId. Refused: ${verdict.refusal}.`,
    );
  const independence = await independentSignerRefusal(target, ctx, verdict.meaning);
  if (independence) throw independence;
  const scope = { sequenceId: seq.id, organizationId: ctx.organizationId, findingId };
  const offered = await shadowFindingDisposable(asQueryable(pool), scope);
  const notOffered = shadowFindingDispositionRefusal(offered, disposition);
  if (notOffered !== null) throw new SubmissionError('SHADOW_FINDING_NOT_DISPOSABLE', notOffered);
  const reason = await dispositionSignatureReason(asQueryable(pool), { organizationId: ctx.organizationId, signatureActionId });
  if (reason === null) {
    throw new SubmissionError('GOVERNED_REQUIRED', 'The signature records no reason, so nothing was closed; sign again with your reason.');

  const runId = Number(offered.runId);


  const judged = offered.judged;
  if (!judged) throw new SubmissionError('SHADOW_FINDING_NOT_DISPOSABLE', offered.reason);
  const update = shadowFindingDispositionUpdate({ ...scope, signerId: ctx.userId, disposition, signatureActionId, reason, judged });
  await applySequenceChangeWithAudit(
    {
      underLock: (client) =>
        recheckUnderSequenceLock(client, {
          sequenceId: seq.id,
          submissionId: seq.submissionId,
          organizationId: ctx.organizationId,
          boundDigest: verdict.boundDigest,
          act: 'disposition',
          recheck: async (q) => {
            const still = shadowFindingDispositionRefusal(await shadowFindingDisposable(q, scope), disposition);
            if (still !== null) throw new SubmissionError('SHADOW_FINDING_NOT_DISPOSABLE', still);
          },
        }),
      text: update.text,
      params: update.params,
      noRowsRefusal: `Finding #${findingId} changed while this disposition was being recorded (it was closed, deleted or moved), so nothing was closed.`,
    },
    {
      organizationId: ctx.organizationId,
      userId: ctx.userId,
      action: SHADOW_FINDING_DISPOSITIONED_ACTION,
      resourceId: seq.id,
      details: { findingId, runId, disposition, signatureActionId, signatureId: verdict.signatureId },
    },
  );
  logger.info('Shadow Review finding closed by signature', { sequenceId: seq.id, findingId, runId, disposition, organizationId: ctx.organizationId });

  const dispositionRecord = await readShadowFindingDisposition(asQueryable(pool), { organizationId: ctx.organizationId, findingId })
    .catch((err: unknown) => {
      logger.warn('A recorded finding disposition could not be read back', {
        sequenceId: seq.id, findingId, err: err instanceof Error ? err.message : String(err),
      });
      return null;
    });
  return { sequenceId: seq.id, findingId, runId, disposition, signatureActionId, signatureId: verdict.signatureId, dispositionRecord };











export type TransmitRoute =
  | { ok: true; gwRegion: string; gwName: string }
  | { ok: false; reason: string };


export function transmitRouteFor(
  region: string,
  clientType: string,
  applicationType?: string | null
): TransmitRoute {
  switch (region) {
    case 'fda':
      return { ok: true, gwRegion: 'fda', gwName: 'esg' };
    case 'jp':
    case 'pmda':
      return { ok: true, gwRegion: 'pmda', gwName: 'pmda_gateway' };
    case 'eu':
    case 'ema': {
      if (clientType === 'mdx' || clientType === 'ivd') return { ok: true, gwRegion: 'ema', gwName: 'eudamed' };
      const type = typeof applicationType === 'string' ? applicationType.trim() : '';
      const entry = type ? resolveToRegistryEntry(type) : null;
      if (!entry) {
        return {
          ok: false,
          reason: `The filing type ${type ? `\"${type}\"` : '(none recorded)'} names no filing in the regulatory registry, so the platform cannot tell whether CESP accepts it (CESP carries national, MRP and DCP filings only; a centralised-procedure filing goes through the EMA eSubmission Gateway / Web Client). Nothing was sent.`,
        };
      }
      const notCesp = cespChannelRefusal(entry, submissionChannelFor(entry));
      if (notCesp) return { ok: false, reason: `${notCesp} Nothing was sent.` };
      return { ok: true, gwRegion: 'ema', gwName: 'cesp' };
    }
    case 'ca':
      return { ok: true, gwRegion: 'ca', gwName: 'hc_cesg' };
    case 'uk':
      return { ok: true, gwRegion: 'uk', gwName: 'mhra_gateway' };
    case 'cn':
      return { ok: true, gwRegion: 'cn', gwName: 'nmpa_gateway' };
    case 'au':
      return { ok: true, gwRegion: 'au', gwName: 'tga_ebs' };
    case 'ch':
      return { ok: true, gwRegion: 'ch', gwName: 'swissmedic_egateway' };
    case 'br':
      return { ok: true, gwRegion: 'br', gwName: 'anvisa_gateway' };
    case 'in':
      return { ok: true, gwRegion: 'in', gwName: 'cdsco_sugam' };
    case 'kr':
      return { ok: true, gwRegion: 'kr', gwName: 'mfds_dbio' };
    case 'sg':
      return { ok: true, gwRegion: 'sg', gwName: 'hsa_prism' };
    default:
      return { ok: false, reason: `No transmit gateway is mapped for region \"${region}\".` };


export function selectGateway(
  region: string,
  clientType: string,
  applicationType?: string | null
): { gwRegion: string; gwName: string } | null {
  const route = transmitRouteFor(region, clientType, applicationType);
  return route.ok ? { gwRegion: route.gwRegion, gwName: route.gwName } : null;



export const TRANSMITTING_STATUS = 'transmitting';

export function resendRefusal(dispatchStatus: string | null | undefined): string | null {
  if (dispatchStatus === TRANSMITTING_STATUS) {
    return `A transmit of this sequence is already in flight (dispatch status '${dispatchStatus}'). It is not sent again while that attempt is unresolved — confirm at the agency whether the package arrived before retrying.`;
  if (dispatchStatus === 'sent' || dispatchStatus === 'acknowledged') {
    return `Sequence was already transmitted (dispatch status '${dispatchStatus}'); it is not sent again. A correction is a new sequence.`;
  return null;

function toDispatchStatus(status: string): 'sent' | 'acknowledged' | 'rejected' {
  if (status === 'rejected' || status === 'validation_failed') return 'rejected';
  if (status === 'ack3_received' || status === 'validation_passed' || status === 'completed') return 'acknowledged';
  return 'sent';

type TransmitClaimExpectation = {
  boundDigest: string; submissionId: number; programId: string | null; applicationId: string;
  fda: boolean; recordedPackageIdentity?: RecordedPackageIdentitySnapshot;
};


function transmitClaimBindingMatches(
  captured: RecordedPackageIdentitySnapshot | undefined,
  digest: string | null,
  expected: TransmitClaimExpectation,
): captured is RecordedPackageIdentitySnapshot {
  return digest === expected.boundDigest && !!captured &&
    captured.identity.programId === expected.programId &&
    captured.identity.applicationNumber === expected.applicationId &&
    sameRecordedPackageIdentitySnapshot(captured, expected.recordedPackageIdentity);


async function claimTransmitSlot(
  sequenceId: number, organizationId: number, signatureActionId: string, attemptId: string,
  expected: TransmitClaimExpectation,
): Promise<{ claimed: false } | { claimed: true; recordedPackageIdentity: RecordedPackageIdentitySnapshot }> {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    await setTenantContextTx(client, organizationId);
    const q = asQueryable(client);
    const parent = await lockSubmissionPackageIdentity(q, organizationId, expected.submissionId);
    const locked = await client.query(`SELECT id, submission_id FROM ectd_sequences WHERE id = $1 AND organization_id = $2 AND deleted_at IS NULL FOR UPDATE`, [sequenceId, organizationId]);
    if (!parent || locked.rows[0]?.submission_id !== expected.submissionId || parent.programId !== expected.programId) {
      throw new SubmissionError('GOVERNED_REQUIRED', 'The sequence parent or the project’s tracking record changed while the transmit claim was being taken. Nothing was assembled or sent; sign the current content again.');
    }


    if (await signatureSpentBy(signatureActionId, `ectd-sequence:${sequenceId}`, organizationId, client)) {
      throw new SubmissionError('GOVERNED_REQUIRED', 'The transmit signature was already used or voided. Sign a new transmit attempt. Nothing was transmitted.');
    }
    await lockRecordedPackageIdentity(q, organizationId, parent.programId);
    const binding = await deriveGovernedTargetBinding(q, `ectd-sequence:${sequenceId}`, organizationId);
    const captured = binding.recordedPackageIdentity;
    if (!transmitClaimBindingMatches(captured, binding.digest, expected)) {
      throw new SubmissionError('GOVERNED_REQUIRED',
        'The sequence leaves or recorded application/applicant facts changed while the transmit claim was being taken. Nothing was assembled or sent; sign the current content again.');
    }
    const missing = packageIdentityRefusal(captured.identity, 'Nothing was sent.', { fda: expected.fda });
    if (missing) throw new SubmissionError('PACKAGE_IDENTITY_MISSING', missing.message);
    const res = await client.query(
      `UPDATE ectd_sequences
          SET dispatch_status = $3, transmit_signature_action_id = $4, transmit_attempt_id = $5, updated_at = NOW()
        WHERE id = $1 AND organization_id = $2 AND deleted_at IS NULL
          AND status = 'dispatched'
          AND (dispatch_status IS NULL OR dispatch_status IN ('pending', 'rejected'))
        RETURNING id`,
      [sequenceId, organizationId, TRANSMITTING_STATUS, signatureActionId, attemptId],
    );
    await client.query('COMMIT');
    return (res.rowCount ?? res.rows?.length ?? 0) > 0
      ? { claimed: true, recordedPackageIdentity: recordedPackageIdentitySnapshot(captured.identity) }
      : { claimed: false };
    await client.query('ROLLBACK');
    throw err;
    client.release();


async function releaseTransmitSlot(sequenceId: number, organizationId: number, signatureActionId: string, attemptId: string): Promise<void> {
  await pool.query(
    `UPDATE ectd_sequences SET dispatch_status = 'pending', transmit_signature_action_id = NULL, transmit_attempt_id = NULL, updated_at = NOW()
      WHERE id = $1 AND organization_id = $2 AND dispatch_status = $3 AND transmit_signature_action_id = $4 AND transmit_attempt_id = $5`,
    [sequenceId, organizationId, TRANSMITTING_STATUS, signatureActionId, attemptId],
  );

    staging: import('../../../shared/schema/gateway-transport').GatewayTransportReadiness | null;
    production: import('../../../shared/schema/gateway-transport').GatewayTransportReadiness | null;
  
  recordedApplicationNumber: string | null;

export function transmitApplicationNumberRefusal(typed: string, recorded: string | null): SubmissionError | null {
  const usable = usableIdentifier('applicationNumber', typed);
  if (usable === null) {
    return new SubmissionError(
      'VALIDATION',
      `\"${typed.trim()}\" is not a usable application number: it must start with a letter or digit and hold only ` +
        'letters, digits, \".\", \"_\" or \"-\" (up to 64). Nothing was sent.',
    );
  if (recorded !== null && usable !== recorded) {
    return new SubmissionError(
      'APPLICATION_NUMBER_MISMATCH',
      `Application number \"${usable}\" does not match the program's recorded application number \"${recorded}\". ` +
        'The record is authoritative: transmit under the recorded number, or correct the project record first. Nothing was sent.',
    );
  return null;


function typedApplicationNumberRefusal(typed: string | undefined, recorded: string | null): string | null {
  if (typeof typed !== 'string' || !typed.trim()) return null;
  return transmitApplicationNumberRefusal(typed, recorded)?.message ?? null;


export async function sequenceTransmitReadiness(
  sequenceId: number,
  ctx: { organizationId: number; userId: number },
  environment?: 'staging' | 'production',
  
  applicationId?: string,
): Promise<SequenceTransmitReadiness> {
  const seq = await getSequence(sequenceId, ctx);
  const submission = await getSubmission(seq.submissionId, ctx);
  const routed = transmitRouteFor(seq.region, submission.clientType, submission.applicationType || seq.type);
  const route: SequenceTransmitReadiness['route'] = routed.ok
    ? { ok: true, region: routed.gwRegion, gateway: routed.gwName }
    : { ok: false, reason: routed.reason };

  const configured: SequenceTransmitReadiness['configured'] = { staging: null, production: null };
  const transportReadiness: SequenceTransmitReadiness['transportReadiness'] = { staging: null, production: null };
  if (routed.ok) {
    const { getGateway, gatewayTransportReadiness } = await import('../submission-gateways/index');

    const gw = getGateway(routed.gwRegion as Parameters<typeof getGateway>[0], routed.gwName as Parameters<typeof getGateway>[1]);
    for (const env of ['staging', 'production'] as const) {
      try {
        configured[env] = await gw.isConfigured(ctx.organizationId, env);
      } catch {
        configured[env] = null;
      }
      transportReadiness[env] = await gatewayTransportReadiness(
        routed.gwRegion as Parameters<typeof gatewayTransportReadiness>[0],
        routed.gwName as Parameters<typeof gatewayTransportReadiness>[1],
        env,
      ).catch(() => null);
    }

  const { assessSequenceDispatchReadiness } = await import('../ectd/assess-dispatch-readiness');
  const assessment = await assessSequenceDispatchReadiness({ sequenceId, organizationId: ctx.organizationId });
  const gate = { cleared: assessment.gate.cleared, blockers: [...assessment.gate.blockers] };


  const identity = await readSubmissionPackageIdentity(pool, ctx.organizationId, submission.id, { region: seq.region });
  const recordedApplicationNumber = identity.applicationNumber;
  const identityRefusal = packageIdentityRefusal(identity, 'Nothing can be sent until it is recorded.', { fda: canonicalRegionOf(seq.region) === 'US' })?.message ?? null;
  const typedNumberRefusal = identityRefusal ?? typedApplicationNumberRefusal(applicationId, recordedApplicationNumber);

  const gatewayRefusal = (): string | null => {
    if (!route.ok) return route.reason;
    const envs = environment ? [environment] : (['staging', 'production'] as const);
    if (envs.every((e) => transportReadiness[e]?.implementation === 'unavailable')) {
      return transportReadiness[envs[0]]?.reason ?? 'This agency transport is unavailable. Nothing can be transmitted.';
    }
    const unusable = envs.filter((e) => configured[e] !== true);
    if (unusable.length < envs.length) return null;
    const unknown = unusable.some((e) => configured[e] === null);
    return unknown
      ? `Whether this sequence's agency gateway is configured for ${envs.join(' or ')} could not be checked, so nothing can be sent until it can.`
      : `This sequence's agency gateway has no ${envs.join(' or ')} credentials configured for this organization, so nothing can be sent. Nothing was signed or transmitted.`;
  const refusal =
    seq.status !== 'dispatched'
      ? `Sequence must be dispatched before transmit (current: ${seq.status}).`
      : resendRefusal(seq.dispatchStatus) ??
        (!gate.cleared ? `Dispatch gate blocks transmit: ${gate.blockers.join(' ')}` : null) ??
        typedNumberRefusal ??
        gatewayRefusal();

  return {
    sequenceStatus: seq.status,
    dispatchStatus: seq.dispatchStatus ?? null,
    route,
    configured,
    transportReadiness,
    recordedApplicationNumber,
    recordedApplicant: identity.applicantName,
    gate,
    refusal,


export async function transmitSequence(params: TransmitSequenceParams): Promise<TransmitSequenceResult> {
  const { sequenceId, ctx, signatureActionId } = params;
  if (params.environment !== 'staging' && params.environment !== 'production') {
    throw new SubmissionError('VALIDATION', 'Transmit requires an explicit environment: staging or production.');
  const environment = params.environment;
  const typed = typeof params.applicationId === 'string' ? params.applicationId.trim() : '';
  if (!typed || /^UNASSIGNED/i.test(typed)) {
    throw new SubmissionError(
      'VALIDATION',
      'Transmit requires the agency application number; a sequence with none recorded is assembled for inspection only, never sent.',
    );
  const unusable = transmitApplicationNumberRefusal(typed, null);
  if (unusable) throw unusable;
  const applicationId = usableIdentifier('applicationNumber', typed) ?? typed;

  const attempt = { reachedGateway: false };
  const result = await voidingSignatureOnRefusal(
    () => transmitDispatchedSequence(params, { environment, applicationId }, attempt),
    { signatureActionId, sequenceId, ctx, step: 'transmit' },
    async (err) => {
      if (!attempt.reachedGateway) return true;
      const { refusedBeforeWire } = await import('../submission-gateways/index');
      return refusedBeforeWire(err);
    },
  );
  if (result.transmitted) return result;
  const voided = await voidRefusedStepSignature(
    signatureActionId, sequenceId, ctx, 'transmit', `Not transmitted: ${result.reason ?? 'the gateway did not send it'}.`,
  );
  return { ...result, signatureVoided: voided === 'voided' };


async function transmitDispatchedSequence(
  params: TransmitSequenceParams,
  checked: { environment: 'staging' | 'production'; applicationId: string },
  attempt: { reachedGateway: boolean },
): Promise<TransmitSequenceResult> {
  const { sequenceId, ctx, signatureActionId } = params;
  const { environment, applicationId } = checked;
  const seq = await getSequence(sequenceId, ctx);
  if (seq.status !== 'dispatched') {
    throw new SubmissionError('INVALID_STATE', `Sequence must be dispatched before transmit (current: ${seq.status}).`);
  const resend = resendRefusal(seq.dispatchStatus);
  if (resend) throw new SubmissionError('INVALID_STATE', resend);


  const submission = await getSubmission(seq.submissionId, ctx);


  const identity = await readSubmissionPackageIdentity(pool, ctx.organizationId, submission.id, { region: seq.region });
  const region = canonicalRegionOf(seq.region);
  if (!region) throw new SubmissionError('GOVERNED_REQUIRED', 'The sequence region is not recognized, so recorded release identity cannot be established. Nothing was sent.');
  const fda = region === 'US';
  const missingIdentity = packageIdentityRefusal(identity, 'Nothing was sent.', { fda });
  if (missingIdentity) throw new SubmissionError('PACKAGE_IDENTITY_MISSING', missingIdentity.message);
  const numberRefusal = transmitApplicationNumberRefusal(applicationId, identity.applicationNumber);
  if (numberRefusal) throw numberRefusal;

  const target = `ectd-sequence:${sequenceId}`;
  const verdict = await governedSignatureVerdict(signatureActionId, target, ctx, 'transmit');
  if (verdict.refusal !== null) {
    throw new SubmissionError(
      'GOVERNED_REQUIRED',
      `A valid e-signature is required: sign ${target} via POST /api/c2c/actions/sign with intent 'transmit', then pass its actionId. Refused: ${verdict.refusal}.`
    );

  const { assessSequenceDispatchReadiness } = await import('../ectd/assess-dispatch-readiness');
  const assessment = await assessSequenceDispatchReadiness({ sequenceId, organizationId: ctx.organizationId });
  if (!assessment.gate.cleared) {
    throw new SubmissionError('DISPATCH_BLOCKED', `Dispatch gate blocks transmit: ${assessment.gate.blockers.join(' ')}`);




  const route = transmitRouteFor(seq.region, submission.clientType, submission.applicationType || seq.type);
  if (!route.ok) {
    throw new SubmissionError('VALIDATION', route.reason);

  const { getGateway, preTransmitFindings } = await import('../submission-gateways/index');

  const gw = getGateway(route.gwRegion as any, route.gwName as any);

  if (!(await gw.isConfigured(ctx.organizationId, environment))) {







    const skipAudit = await recordAuditRow({
      organizationId: ctx.organizationId,
      userId: ctx.userId,
      action: 'ECTD_TRANSMIT_SKIPPED',
      resourceType: 'ectd_sequence',
      resourceId: sequenceId,
      details: { region: seq.region, gateway: route.gwName, reason: 'gateway_not_configured', environment },
    });
    return {
      transmitted: false,
      reason: 'gateway_not_configured',
      skipAudit,
      region: seq.region,
      gateway: route.gwName,
      dispatchStatus: seq.dispatchStatus ?? 'pending',
    };

  
  const attemptId = randomUUID();
  const claim = await claimTransmitSlot(sequenceId, ctx.organizationId, signatureActionId, attemptId,
    { boundDigest: verdict.boundDigest, submissionId: seq.submissionId, programId: submission.programId ?? null, applicationId, fda, recordedPackageIdentity: verdict.recordedPackageIdentity });
  if (!claim.claimed) {
    const fresh = await getSequence(sequenceId, ctx);
    throw new SubmissionError(
      'INVALID_STATE',
      resendRefusal(fresh.dispatchStatus) ??
        `Sequence ${sequenceId} is no longer in a transmittable state (dispatch status '${fresh.dispatchStatus ?? 'unknown'}').`,
    );



  const { assembleSequence } = await import('../ectd/assemble-from-core');
  let assembled;
  try {
    const capturedIdentity = claim.recordedPackageIdentity.identity;



    const nonFdaApplicantId = params.sponsorId === undefined ? '' : usableIdentifier('applicantId', params.sponsorId);
    if (!fda && nonFdaApplicantId === null) throw new SubmissionError('VALIDATION', 'The applicant id is not a usable regional identifier. Nothing was sent.');
    assembled = await assembleSequence({
      sequenceId,
      organizationId: ctx.organizationId,
      userId: ctx.userId,
      applicationId: capturedIdentity.applicationNumber as string,
      sponsorId: fda ? capturedIdentity.applicantId as string : nonFdaApplicantId as string,
      sponsorName: capturedIdentity.applicantName as string,
      recordedPackageIdentity: claim.recordedPackageIdentity,
      ...(fda ? { contacts: fdaContactsForPackageIdentity(capturedIdentity) } : {}),
    });
    await releaseTransmitSlot(sequenceId, ctx.organizationId, signatureActionId, attemptId);
    throw err;











  const unresolved = assembled.unresolvedLeaves;
  if (unresolved.length > 0) {
    await assembled.cleanup();
    const isExternal = (l: { documentTable: string | null }) =>
      externalDocumentTableReason(l.documentTable) !== null;
    const defects = unresolved.filter((l) => !isExternal(l));
    const external = unresolved.filter(isExternal);
    const parts: string[] = [];
    if (defects.length > 0) {
      parts.push(
        `${defects.length} reference a document that could not be assembled ` +
          `(${defects.map((d) => `${d.documentTable}:${d.documentId}`).join(', ')})`,
      );
    }
    if (external.length > 0) {
      parts.push(
        `${external.length} external-storage document(s) (vault/onboarding) whose bytes were not materialized into the package ` +
          `(${external.map((d) => `${d.documentTable}:${d.documentId}`).join(', ')})`,
      );
    }
    await releaseTransmitSlot(sequenceId, ctx.organizationId, signatureActionId, attemptId);
    throw new SubmissionError(
      'DISPATCH_BLOCKED',
      `Transmit blocked — the transmitted sequence would be missing ${unresolved.length} leaf file(s): ${parts.join('; ')}. ` +
        `Every eCTD leaf must be physically present in the package.`,
    );







  const { assembledTransmitBlockers } = await import('../ectd/assemble-from-core');
  const gaps = assembledTransmitBlockers(assembled);
  if (gaps.length > 0) {
    await assembled.cleanup();
    await releaseTransmitSlot(sequenceId, ctx.organizationId, signatureActionId, attemptId);
    throw new SubmissionError(
      'DISPATCH_BLOCKED',
      `Transmit blocked — ${gaps.join('; ')}. A transmitted sequence must carry every placed leaf, and only approved documents.`,
    );



  try {
    const current = await deriveGovernedTargetBinding(asQueryable(pool), target, ctx.organizationId);
    if (current.digest !== verdict.boundDigest ||
      !sameRecordedPackageIdentitySnapshot(assembled.recordedPackageIdentity, claim.recordedPackageIdentity) ||
      !sameRecordedPackageIdentitySnapshot(current.recordedPackageIdentity, claim.recordedPackageIdentity)) {
      throw new SubmissionError('GOVERNED_REQUIRED',
        'The assembled package does not match the approved leaves and recorded application/applicant facts at the transmit checkpoint. Nothing was sent; review the record and sign again.');
    }
    await assembled.cleanup();
    await releaseTransmitSlot(sequenceId, ctx.organizationId, signatureActionId, attemptId);
    throw err;

  let result;
  try {
    attempt.reachedGateway = true;
    result = await gw.transmit({
      organizationId: ctx.organizationId,
      userId: ctx.userId,



      programId: submission.programId ?? null,
      packageId: null,
      bundle: assembled.bundle,
      environment,
      submissionType: seq.type ?? undefined,
      metadata: { applicationId: claim.recordedPackageIdentity.identity.applicationNumber, sequence: seq.sequenceNumber, environment,
        recordedPackageIdentity: claim.recordedPackageIdentity },

      dispatchBinding: { attemptId, organizationId: ctx.organizationId, sequenceId, signatureActionId, bundleSha256: assembled.bundle.sha256 },



      authorization: {
        kind: 'governed-signature',
        signatureActionId,
        actorUserId: ctx.userId,
      },
    });




    const { refusedBeforeWire } = await import('../submission-gateways/index');
    if (refusedBeforeWire(err)) await releaseTransmitSlot(sequenceId, ctx.organizationId, signatureActionId, attemptId);
    throw err;



    await assembled.cleanup();

  const dispatchStatus = toDispatchStatus(result.status);



  const preTransmit = preTransmitFindings(result);
  await applySequenceChangeWithAudit(
    {





      text: `UPDATE ectd_sequences SET dispatch_status = $1, updated_at = NOW() WHERE id = $2 AND organization_id = $3 AND dispatch_status = $4 AND transmit_signature_action_id = $5 AND transmit_attempt_id = $6`,
      params: [dispatchStatus, sequenceId, ctx.organizationId, TRANSMITTING_STATUS, signatureActionId, attemptId],
      noRowsRefusal:
        `Sequence ${sequenceId} was no longer the in-flight transmit when the result came back; ` +
        `its dispatch status was resolved by something else. The package WAS handed to the gateway — ` +
        `confirm at the agency before any further action.`,
    },
    {
      organizationId: ctx.organizationId,
      userId: ctx.userId,
      action: 'ECTD_TRANSMITTED',
      resourceId: sequenceId,
      details: {
        region: seq.region,
        gateway: route.gwName,
        transmittalId: result.transmittalId,
        transmissionId: result.transmissionId ?? null,
        packageSha256: assembled.bundle.sha256,
        recordedPackageIdentity: claim.recordedPackageIdentity,
        status: result.status,
        environment,
        signatureActionId,


        preTransmitFailedChecks: preTransmit.failedChecks,
        preTransmitWarnings: preTransmit.warnings,


        sentPackage: result.sentPackage ?? null,

        ...withWaiverReliedOn(assessment),
      },
    },
  );
  logger.info('Transmitted sequence to agency gateway', { sequenceId, region: seq.region, gateway: route.gwName, status: result.status });

  return {
    transmitted: true,
    region: seq.region,
    gateway: route.gwName,
    transmittalId: result.transmittalId,
    transmissionId: result.transmissionId,
    status: result.status,
    dispatchStatus,
    preTransmitFailedChecks: preTransmit.failedChecks,
    preTransmitWarnings: preTransmit.warnings,


export async function listLeaves(
  sequenceId: number,
  ctx: { organizationId: number }
): Promise<SubmissionLeaf[]> {
  await getSequence(sequenceId, ctx);
  const rows = await db
    .select()
    .from(submissionLeaves)
    .where(
      and(
        eq(submissionLeaves.sequenceId, sequenceId),
        eq(submissionLeaves.organizationId, ctx.organizationId),
        isNull(submissionLeaves.deletedAt)
      )
    )
    .orderBy(submissionLeaves.sectionCode);
  return rows as SubmissionLeaf[];
