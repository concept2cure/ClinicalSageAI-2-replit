function owner(el: Element): string {
  const key = Object.keys(el).find((k) => k.startsWith('__reactFiber$'));
  let f: any = key ? (el as any)[key] : null; const names: string[] = [];
  while (f && names.length < 4) { const t = f.type; if (typeof t === 'function' && t.name) names.push(t.name); else if (t && typeof t === 'object' && (t.displayName || t.render?.name)) names.push(t.displayName || t.render.name); f = f.return; }
  return names.join(' < ');
}
function dump(tag: string, re: RegExp) {
  const out: string[] = [];
  for (const b of Array.from(document.querySelectorAll('button, [role=button]'))) {
    const name = computeAccessibleName(b as Element);
    if (!re.test(name)) continue;
    const grp = (b as Element).closest('[role=group]');
    out.push(`  name=${JSON.stringify(name)} testid=${b.getAttribute('data-testid')} aria-label=${JSON.stringify(b.getAttribute('aria-label'))} title=${JSON.stringify(b.getAttribute('title'))} disabled=${(b as HTMLButtonElement).disabled} group=${JSON.stringify(grp?.getAttribute('aria-label') ?? null)} class=${JSON.stringify(b.getAttribute('class'))} owner=${owner(b as Element)}`);
  }
  console.log(`PROBE[${tag}] ${re} matched ${out.length}\n${out.join('\n')}`);
}
