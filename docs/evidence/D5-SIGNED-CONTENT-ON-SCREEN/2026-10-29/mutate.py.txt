M = [

    ('M2', 'A facet this client cannot read is taken as unsigned, not unavailable', SCS,

    ('M3', 'A failed record read is shown as unsigned', SCS,
     "      : { reading: false, facet: UNAVAILABLE_FACET, record: null };",

def sha(path):
    with open(path, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def run(tests):
    if os.path.exists(TMP):
        os.remove(TMP)
    cmd = ['timeout', '400', 'npx', 'vitest', 'run', *tests, '--reporter=json', '--outputFile=' + TMP]
    subprocess.run(cmd, cwd=ROOT, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    try:
        with open(TMP) as f:
            r = json.load(f)
    except (OSError, ValueError):
        return None
    failed = []
    for file in r.get('testResults', []):
        name = os.path.relpath(file.get('name', ''), ROOT)
        assertions = file.get('assertionResults', [])
        if not assertions and file.get('status') == 'failed':
            failed.append(f'{name}: the file failed to load')
        for a in assertions:
            if a.get('status') == 'failed':
                failed.append(f"{os.path.basename(name)} > {a.get('fullName') or a.get('title')}")
    return r.get('numFailedTests', 0), r.get('numTotalTests', 0), failed


files = sorted({m[2] for m in M})
before = {p: sha(p) for p in files}
lines = [f"# Mutations: each applied to product code as an exact replacement (its old string occurs once), the named tests run",
         f"# (npx vitest run <tests> --reporter=json), then the file restored from the bytes read before it, and its sha256 checked.",
         f"# Working tree at {subprocess.run(['git', 'rev-parse', '--short', 'HEAD'], capture_output=True, text=True).stdout.strip()} + this change; {time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())}.",
         '# sha256 before:'] + [f'#   {before[p]}  {p}' for p in files] + ['']
red = 0
for mid, desc, path, old, new, tests in M:
    with open(path, 'rb') as f:
        original = f.read()
    text = original.decode('utf8')
    n = text.count(old)
    if n != 1:
        lines.append(f'{mid}  NOT APPLIED: the old string occurs {n} times in {path}')
        continue
    with open(path, 'wb') as f:
        f.write(text.replace(old, new, 1).encode('utf8'))
    try:
        result = run(tests)
    finally:
        with open(path, 'wb') as f:
            f.write(original)
    restored = sha(path) == before[path]
    if result is None:
        lines.append(f'{mid}  {desc}\n     {path}\n     NO RESULT (the run produced no report); restored: {restored}')
        continue
    nf, nt, failed = result
    red += 1 if nf > 0 or any('failed to load' in x for x in failed) else 0
    lines.append(f'{mid}  {desc}\n     {path}\n     tests: {", ".join(os.path.basename(t) for t in tests)}\n'
                 f'     RED {nf} of {nt}; restored and sha256 equal: {restored}')
    for x in failed[:8]:
        lines.append(f'       - {x}')
    if len(failed) > 8:
        lines.append(f'       - … and {len(failed) - 8} more')
    lines.append('')
    print(mid, nf, nt, restored, flush=True)

after = {p: sha(p) for p in files}
lines += [f'# {red} of {len(M)} mutations turned a test red.', '# sha256 after (each equal to before):']
lines += [f'#   {after[p]}  {p}  {"equal" if after[p] == before[p] else "DIFFERENT"}' for p in files]
with open(OUT, 'w') as f:
    f.write('\n'.join(lines) + '\n')
print('done', red, len(M))
