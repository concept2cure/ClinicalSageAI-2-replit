  command = apply
  assert {
    condition     = contains([for e in output.b9_api_container_environment : "${e.name}=${e.value}"], "TRUST_PROXY_HOPS=2")
    error_message = "The API task definition must carry TRUST_PROXY_HOPS=2."
  }
}
