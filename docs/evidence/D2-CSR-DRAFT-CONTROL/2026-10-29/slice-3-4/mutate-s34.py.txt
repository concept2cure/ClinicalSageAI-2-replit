sha = lambda b: hashlib.sha256(b).hexdigest()
only = set(sys.argv[1:])
results = []
for name, rel, old, new, tests in M:
    if only and name.split()[0] not in only: continue
    path = os.path.join(ROOT, rel)
    original = open(path, 'rb').read()
    src = original.decode()
    assert src.count(old) == 1, (name, 'old', src.count(old))
    mutated = src.replace(old, new)
    assert mutated.count(new) == 1, (name, 'new', mutated.count(new))
    open(path, 'w').write(mutated)
    mutated_b = mutated.encode()
    t0 = time.time()
    try:
        r = subprocess.run(['npx', 'vitest', 'run', '--config', 'vitest.config.ts', *tests], cwd=ROOT, capture_output=True, text=True,
                           timeout=900, env={**os.environ, 'NODE_OPTIONS': '--max-old-space-size=3072'})
        out = r.stdout + r.stderr
        failed = [l.strip() for l in r.stdout.splitlines() if l.strip().startswith('×')]
        code = r.returncode
    finally:
        now = open(path, 'rb').read()
        if now == mutated_b:
            open(path, 'wb').write(original); how = 'original bytes written back'
        else:
            cur = now.decode(); assert cur.count(new) == 1, (name, 'concurrent edit lost the mutation marker')
            open(path, 'w').write(cur.replace(new, old)); how = 'mutated text reversed in place (file changed by another writer meanwhile)'
    restored = open(path, 'rb').read()
    results.append({'mutation': name, 'file': rel, 'tests': tests, 'exitCode': code, 'killed': code != 0 and bool(failed),
                    'failingTests': failed, 'seconds': round(time.time() - t0, 1), 'sha256Before': sha(original),
                    'sha256Restored': sha(restored), 'restoredByteForByte': restored == original, 'restore': how})
    print(json.dumps({k: results[-1][k] for k in ('mutation', 'killed', 'restoredByteForByte')}), len(failed), flush=True)
tag = '-'.join(sorted(only)) if only else 'all'
json.dump(results, open(os.path.join(OUT, f'mutations-{tag}.json'), 'w'), indent=1)
print('killed', sum(x['killed'] for x in results), 'of', len(results), '; restored byte for byte', sum(x['restoredByteForByte'] for x in results))
sys.exit(0 if all(x['killed'] and x['restoredByteForByte'] for x in results) else 1)
