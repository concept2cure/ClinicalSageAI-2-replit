// @vitest-environment jsdom
/** Reporting reaches the canonical paged inventory without losing explicit choices. */
import React from 'react';
import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest';
import { act, cleanup, fireEvent, render, screen, waitFor, within } from '@testing-library/react';
import type { CompanyLegalDocument } from '@shared/company-documents';
import { endCompanySession, seedCompanySession } from './_company-session-fixture';

const apiRequest = vi.hoisted(() => vi.fn());
const downloadBlob = vi.hoisted(() => vi.fn(() => true));
vi.mock('@/lib/queryClient', async original => ({ ...(await original<typeof import('@/lib/queryClient')>()), apiRequest }));
vi.mock('../download', () => ({ downloadBlob }));
import { InsightsLegalPackage } from '../surfaces/InsightsLegalPackage';
import { PagedCompanyLegalPicker } from '../surfaces/company-documents/PagedCompanyLegalPicker';

const PATH = '/api/client-branding/company-documents';
const BUNDLE = '33333333-3333-4333-8333-333333333333';
const ZIP = new Blob(['retained-server-report-archive'], { type: 'application/zip' });
const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { 'Content-Type': 'application/json' } });
function original(index: number): CompanyLegalDocument {
  return { id: `legal-${index}`, title: `Authorization ${String(index).padStart(2, '0')}`, kind: 'authorization',
    programId: '11111111-1111-4111-8111-111111111111', vaultDocumentId: '22222222-2222-4222-8222-222222222222',
    contentHash: 'a'.repeat(64), version: '3', agencies: ['FDA'], countries: ['US'], submissionTypes: ['ind'], language: 'en',
    effectiveOn: null, expiresOn: null, active: true, registeredAt: '2026-10-01T00:00:00Z', registeredBy: '42', transmitRefusal: null };
}
let documents: CompanyLegalDocument[];
let token: string;
function wire() {
  apiRequest.mockImplementation(async (method: string, url: string) => {
    if (method === 'POST') return new Response(ZIP, { headers: { 'Content-Type': 'application/zip' } });
    if (url.includes('/designs?')) return json({ data: [], nextBefore: null });
    // Keep the old transport valid: RED must expose missing paging, not a fixture refusal.
    if (url === PATH) return json({ data: { legalDocuments: documents } });
    if (url.startsWith(`${PATH}/legal-documents?`)) {
      const params = new URL(url, 'http://fixture').searchParams;
      const offset = Number(params.get('offset') ?? 0), limit = Number(params.get('limit') ?? 25);
      const matching = documents.filter(document => (!params.get('q') || document.title.toLowerCase().includes(params.get('q')!.toLowerCase()))
        && (!params.get('kind') || document.kind === params.get('kind')));
      return json({ data: matching.slice(offset, offset + limit), page: { total: matching.length, availableTotal: documents.length,
        offset, limit, hasMore: offset + limit < matching.length, nextOffset: offset + limit < matching.length ? offset + limit : null },
      facets: { agencies: ['FDA'], languages: ['en'] } });
    }
    throw new Error(`Unexpected fixture request: ${method} ${url}`);
  });
}
const posts = () => apiRequest.mock.calls.filter(([method]) => method === 'POST');
const reads = () => apiRequest.mock.calls.filter(([method, url]) => method === 'GET' && url.startsWith(`${PATH}/legal-documents?`));
const lastQuery = () => new URL(String(reads().at(-1)?.[1]), 'http://fixture').searchParams;
async function openChooser() {
  render(<InsightsLegalPackage bundleId={BUNDLE} disabled={false} />);
  fireEvent.click(screen.getByRole('button', { name: /Choose legal companions/ }));
  await screen.findByRole('checkbox', { name: /Authorization 01/ });
}
function selectPage() { screen.getAllByRole('checkbox').forEach(checkbox => fireEvent.click(checkbox)); }
beforeEach(() => {
  localStorage.clear(); sessionStorage.clear(); token = seedCompanySession();
  documents = Array.from({ length: 26 }, (_, index) => original(index + 1));
  apiRequest.mockReset(); downloadBlob.mockReset().mockReturnValue(true); wire();
});
afterEach(() => { cleanup(); endCompanySession(); vi.unstubAllGlobals(); });

describe('Reporting legal chooser reaches paged exact-version selection', () => {
  it('keeps the first page choice while adding an original from the next page, then posts only those IDs', async () => {
    render(<InsightsLegalPackage bundleId={BUNDLE} disabled={false} />);
    fireEvent.click(screen.getByRole('button', { name: /Choose legal companions/ }));
    fireEvent.click(await screen.findByRole('checkbox', { name: /Authorization 01/ }));
    fireEvent.click(screen.getByRole('button', { name: 'Next legal versions' }));
    fireEvent.click(await screen.findByRole('checkbox', { name: /Authorization 26/ }));
    fireEvent.click(screen.getByRole('button', { name: /Download reports and 2/ }));
    await waitFor(() => expect(downloadBlob).toHaveBeenCalledOnce());
    expect(posts()).toEqual([['POST', `/api/report-os/bundles/${BUNDLE}/export.zip`, { legalDocumentIds: ['legal-1', 'legal-26'] }]]);
    expect(apiRequest.mock.calls.some(([, url]) => url === PATH)).toBe(false);
  });

  it('opens only an explicit bounded page and leaves every original unselected', async () => {
    render(<InsightsLegalPackage bundleId={BUNDLE} disabled={false} />);
    expect(apiRequest).not.toHaveBeenCalled();
    fireEvent.click(screen.getByRole('button', { name: /Choose legal companions/ }));
    await screen.findByRole('checkbox', { name: /Authorization 01/ });
    expect(screen.getAllByRole('checkbox')).toHaveLength(25);
    expect(screen.getAllByRole('checkbox').every(checkbox => !(checkbox as HTMLInputElement).checked)).toBe(true);
    expect(lastQuery().get('limit')).toBe('25'); expect(lastQuery().get('offset')).toBe('0');
    expect(lastQuery().get('includeArchived')).toBe('false');
    expect(apiRequest.mock.calls.some(([, url]) => url === PATH)).toBe(false);
    expect(posts()).toHaveLength(0);
  });

  it('keeps an aggregate fifty-ID bound across three pages, search and kind filtering while allowing off-page removal', async () => {
    documents = Array.from({ length: 51 }, (_, index) => ({ ...original(index + 1), kind: index === 50 ? 'cover_letter' : 'authorization' }));
    await openChooser(); selectPage();
    fireEvent.click(screen.getByRole('button', { name: 'Next legal versions' }));
    await screen.findByRole('checkbox', { name: /Authorization 26/ }); selectPage();
    expect(screen.getByRole('button', { name: /Download reports and 50/ })).toBeTruthy();
    fireEvent.click(screen.getByRole('button', { name: 'Next legal versions' }));
    const extra = await screen.findByRole('checkbox', { name: /Authorization 51/ });
    expect((extra as HTMLInputElement).disabled).toBe(true);
    fireEvent.change(screen.getByRole('textbox', { name: /^Search / }), { target: { value: 'Authorization 51' } });
    fireEvent.click(screen.getByRole('button', { name: 'Search legal versions' }));
    await waitFor(() => expect(lastQuery().get('q')).toBe('Authorization 51'));
    expect((await screen.findByRole('checkbox', { name: /Authorization 51/ }) as HTMLInputElement).disabled).toBe(true);
    fireEvent.change(screen.getByRole('combobox', { name: /Document kind/ }), { target: { value: 'cover_letter' } });
    await waitFor(() => expect(lastQuery().get('kind')).toBe('cover_letter'));
    expect((await screen.findByRole('checkbox', { name: /Authorization 51/ }) as HTMLInputElement).disabled).toBe(true);
    const offPage = screen.getByText(/^Selected: legal-1\s*$/);
    fireEvent.click(within(offPage).getByRole('button', { name: 'Remove selected legal version' }));
    const available = screen.getByRole('checkbox', { name: /Authorization 51/ });
    expect((available as HTMLInputElement).disabled).toBe(false); fireEvent.click(available);
    fireEvent.click(screen.getByRole('button', { name: /Download reports and 50/ }));
    await waitFor(() => expect(posts()).toHaveLength(1));
    expect(posts()[0][2]).toEqual({ legalDocumentIds: Array.from({ length: 50 }, (_, index) => `legal-${index + 2}`) });
    expect(new Set(posts()[0][2].legalDocumentIds).size).toBe(50);
  });

  it('shows expiry, revocation and current-source refusals without offering those originals for selection', async () => {
    documents = [{ ...original(1), expiresOn: '2000-01-01' },
      { ...original(2), workflow: { custodianUserId: null, reminderDays: [], replacementId: null,
        revokedAt: '2026-10-20T00:00:00Z', revokedBy: '42', revocationReason: 'Authority withdrawn', authority: null,
        signatureEvidence: null, translationOfId: null, certifiedTranslationEvidence: null,
        updatedAt: '2026-10-20T00:00:00Z', updatedBy: '42' } },
      { ...original(3), transmitRefusal: 'The recorded source is no longer approved.' }];
    await openChooser();
    expect(screen.getAllByRole('checkbox').every(checkbox => (checkbox as HTMLInputElement).disabled)).toBe(true);
    expect(screen.getByText(/document has expired/)).toBeTruthy();
    expect(screen.getByText(/document has been revoked/)).toBeTruthy();
    expect(screen.getByText(/source is no longer approved/)).toBeTruthy();
    expect(posts()).toHaveLength(0);
  });

  it('clears the selection and withdraws cached rows while an explicit refresh reads a fresh visit', async () => {
    await openChooser(); fireEvent.click(screen.getByRole('checkbox', { name: /Authorization 01/ }));
    const normal = apiRequest.getMockImplementation()!;
    let finish!: (value: Response) => void;
    apiRequest.mockImplementation((method: string, url: string) => method === 'GET' && url.startsWith(`${PATH}/legal-documents?`)
      ? new Promise<Response>(resolve => { finish = resolve; }) : normal(method, url));
    fireEvent.click(screen.getByRole('button', { name: /Refresh legal companions/ }));
    await waitFor(() => expect(finish).toBeTypeOf('function'));
    expect(screen.queryByRole('checkbox')).toBeNull();
    expect((screen.getByRole('button', { name: /Download reports and 0/ }) as HTMLButtonElement).disabled).toBe(true);
    await act(async () => { finish(json({ data: [original(2)], page: { total: 1, availableTotal: 1, limit: 25, offset: 0, hasMore: false, nextOffset: null } })); });
    expect((await screen.findByRole('checkbox', { name: /Authorization 02/ }) as HTMLInputElement).checked).toBe(false);
    expect(posts()).toHaveLength(0);
  });

  it('retries the same failed page without discarding an off-page selection or exporting cached rows', async () => {
    await openChooser(); fireEvent.click(screen.getByRole('checkbox', { name: /Authorization 01/ }));
    const normal = apiRequest.getMockImplementation()!; let fail = true;
    apiRequest.mockImplementation((method: string, url: string) => method === 'GET' && url.includes('offset=25') && fail
      ? Promise.resolve(json({ message: 'The legal page is temporarily unavailable.' }, 503)) : normal(method, url));
    fireEvent.click(screen.getByRole('button', { name: 'Next legal versions' }));
    await screen.findByText('The legal page is temporarily unavailable.');
    expect(screen.queryByRole('checkbox')).toBeNull();
    fail = false; fireEvent.click(screen.getByRole('button', { name: 'Retry legal version search' }));
    fireEvent.click(await screen.findByRole('checkbox', { name: /Authorization 26/ }));
    expect(lastQuery().get('offset')).toBe('25');
    fireEvent.click(screen.getByRole('button', { name: /Download reports and 2/ }));
    await waitFor(() => expect(posts()).toHaveLength(1));
    expect(posts()[0][2]).toEqual({ legalDocumentIds: ['legal-1', 'legal-26'] });
  });

  it.each([
    ['a missing page', { data: [original(1)], page: null }],
    ['a null record', { data: [null], page: { total: 1, availableTotal: 1, limit: 25, offset: 0, hasMore: false, nextOffset: null } }],
  ])('reports an unverifiable inventory response with %s without presenting it as an empty library', async (_name, body) => {
    apiRequest.mockResolvedValue(json(body));
    render(<InsightsLegalPackage bundleId={BUNDLE} disabled={false} />);
    fireEvent.click(screen.getByRole('button', { name: /Choose legal companions/ }));
    expect(await screen.findByText(/could not be verified|could not be loaded|unverified/i)).toBeTruthy();
    expect(screen.queryByRole('checkbox')).toBeNull();
    expect(screen.queryByText(/No accessible registered legal documents/)).toBeNull();
  });
});

describe('optional reporting picker policy preserves ordinary mapping defaults', () => {
  it('rechecks a refusal in a retained row handler and still allows removal of a selected refused ID', async () => {
    documents = [original(1)]; const change = vi.fn(); let refused = false;
    const view = render(<PagedCompanyLegalPicker token={token} selected={[]} onChange={change} multiple disabled={false}
      label="Policy originals" rowRefusal={() => refused ? 'This authority was withdrawn.' : null} maxSelected={50} />);
    const checkbox = await screen.findByRole('checkbox', { name: /Authorization 01/ });
    refused = true; fireEvent.click(checkbox);
    expect(change).not.toHaveBeenCalled();
    view.rerender(<PagedCompanyLegalPicker token={token} selected={['legal-1']} onChange={change} multiple disabled={false}
      label="Policy originals" rowRefusal={() => 'This authority was withdrawn.'} maxSelected={50} />);
    fireEvent.click(screen.getByRole('button', { name: 'Remove selected legal version' }));
    expect(change).toHaveBeenCalledWith([]);
  });

  it('keeps review-needed versions selectable when a mapping caller supplies neither optional policy', async () => {
    documents = [{ ...original(1), transmitRefusal: 'Approval review is still needed.' }];
    const change = vi.fn();
    render(<PagedCompanyLegalPicker token={token} selected={[]} onChange={change} multiple disabled={false} label="Ordinary mappings" />);
    const checkbox = await screen.findByRole('checkbox', { name: /Authorization 01/ });
    expect((checkbox as HTMLInputElement).disabled).toBe(false); fireEvent.click(checkbox);
    expect(change).toHaveBeenCalledWith(['legal-1']);
  });
});
