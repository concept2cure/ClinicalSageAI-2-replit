BEGIN;
-- Two throwaway orgs (the sync trigger mirrors them into identity.organizations),
-- one regulatory program and one vault document for B, as the owner.

SELECT uuid::text AS a FROM organizations WHERE id=90391 \gset
SELECT uuid::text AS b FROM organizations WHERE id=90392 \gset

  VALUES ('probe-B-program','PROBE-B',90392,'IND','drug','FDA','probe') RETURNING id::text AS bprog \gset

  VALUES (:'bprog', 90392, repeat('b',64), 'B-CONFIDENTIAL-CMC', 'b.txt', 'completed', 'txt');

SELECT set_config('app.rls_enforce','on',true), set_config('app.current_org_id', :'a', true), set_config('app.current_tenant_id','90391',true);
\echo '-- 1. tenant A reads B''s vault before (expect 0):'

  VALUES (:'b', :'a', CURRENT_DATE, true);

SELECT document_title AS a_now_reads FROM vault.documents WHERE document_title='B-CONFIDENTIAL-CMC';
\echo '-- 4. tenant A overwrites B''s document title:'

SELECT document_title AS b_sees FROM vault.documents WHERE organization_id=90392;
ROLLBACK;
