// Running-app walk of the P-95 artifact review step (ConversationArtifactReview.tsx).
// Real server (port 5055) on a local, fully migrated database; real dev-login sessions.
// The ONLY substituted piece: the AnA stream response that announces a draft (no model key here).
// It names an artifact that genuinely exists in the database, so every control after it hits the real server.
const { chromium } = require('playwright');
const fs = require('fs');
const path = require('path');

const BASE = 'http://127.0.0.1:5000';
const OUT = __dirname;
const ARTIFACT = 'art_p95_walk_v2';
const TITLE = 'P-95 walk 2 — 510(k) Summary';
const log = [];
const note = (s) => { log.push(s); console.log(s); };

async function session(email) {
  const dl = await (await fetch(`${BASE}/api/auth/dev-login`, { method: 'POST',
    headers: { 'Content-Type': 'application/json', Origin: BASE }, body: JSON.stringify({ email }) })).json();
  if (!dl.accessToken) throw new Error('dev-login failed for ' + email + ': ' + JSON.stringify(dl).slice(0, 300));
  const sess = await (await fetch(`${BASE}/api/v1/auth/session`, { headers: { Authorization: `Bearer ${dl.accessToken}`, Origin: BASE } })).json();
  if (!sess.authenticated) throw new Error('session invalid for ' + email);
  return { token: dl.accessToken, refresh: dl.refreshToken, user: sess.user };
}

async function page(browser, s) {
  const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 }, deviceScaleFactor: 2 });
  const expiry = new Date(Date.now() + 23 * 3600e3).toISOString();
  await ctx.addInitScript(([t, r, e, u]) => {
    for (const st of [window.localStorage, window.sessionStorage]) {
      st.setItem('trialsage_access_token', t); st.setItem('trialsage_refresh_token', r);
      st.setItem('trialsage_token_expiry', e); st.setItem('trialsage_user', u);
    }
  }, [s.token, s.refresh, expiry, JSON.stringify(s.user)]);
  const p = await ctx.newPage();
  p.on('response', async (res) => {
    const u = res.url();
    if (/\/api\/concept2cure\/projects\/[^/]+\/(team|artifacts\/[^/]+\/(status|reviewers|reviews\/(status|submit)))/.test(u)) {
      const h = res.headers();
      let meta = '';
      if (res.request().method() !== 'GET') { try { const b = await res.json(); meta = ' meta.auditTrail=' + JSON.stringify(b?.meta?.auditTrail ?? null) + (b?.error ? ' error=' + JSON.stringify(b.error).slice(0, 200) : ''); } catch {} }
      note(`  HTTP ${res.request().method()} ${u.replace(BASE, '')} -> ${res.status()} audit-persisted=${h['x-audit-row-persisted'] ?? '-'} chained=${h['x-audit-row-chained'] ?? '-'}${meta}`);
    }
  });
  return { ctx, p };
}

const shot = async (p, name) => { await p.screenshot({ path: path.join(OUT, name), fullPage: false }); note(`  screenshot ${name}`); };

(async () => {
  const browser = await chromium.launch();
  try {
    // ── 1. The author (admin, the artifact's creator) in the conversation ──
    const admin = await session('e2e-admin@trialsage.test');
    note(`admin session: user ${admin.user.id}, org ${admin.user.organizationId}`);
    const { p } = await page(browser, admin);
    await p.goto(`${BASE}/concept2cure/projects`, { waitUntil: 'domcontentloaded' });
    await p.getByText('P-95 review walk').first().waitFor({ timeout: 120000 });
    await p.getByText('P-95 review walk').first().click();
    await p.waitForTimeout(3000);
    note(`project opened: ${JSON.stringify(await p.evaluate(() => window.C2C_PROJECT ? { id: window.C2C_PROJECT.id, title: window.C2C_PROJECT.title || window.C2C_PROJECT.name } : null))}`);

    // The stream announces AnA's draft of the existing artifact (substituted: no model key in this environment).
    await p.route('**/api/ana-ri/stream', async (route) => {
      const ev = (o) => `data: ${JSON.stringify(o)}\n\n`;
      await route.fulfill({ status: 200, headers: { 'content-type': 'text/event-stream' }, body:
        ev({ type: 'text', content: 'Here is the 510(k) draft for review.' }) +
        ev({ type: 'artifact_draft', title: TITLE, content: '# SYNTHETIC TEST DATA\n\nDraft body.', documentType: '510(k) summary' }) +
        ev({ type: 'artifact_version_saved', artifactId: ARTIFACT, version: 1, title: TITLE }) +
        ev({ type: 'done', provider: 'walk-substituted-stream' }) });
    });
    await p.goto(`${BASE}/concept2cure/conversation-thread`, { waitUntil: 'domcontentloaded' });
    const composer = p.locator('textarea').first();
    await composer.waitFor({ timeout: 120000 });
    await composer.fill('Draft the 510(k) summary for review.');
    await composer.press('Enter');
    await p.getByText(TITLE).first().waitFor({ timeout: 60000 });
    note('artifact card shown in the panel');
    await p.locator('.ct-art-head', { hasText: TITLE }).first().click();
    await shot(p, '01-card-draft.png');

    await p.getByRole('button', { name: new RegExp('Route ' + TITLE.replace(/[()]/g, '.') + ' for review') }).click();
    await p.getByText('In review').first().waitFor({ timeout: 30000 });
    note('status: In review');
    await p.getByText(/No reviewer is named yet|Waiting for a decision|could not be read/).first().waitFor({ timeout: 30000 });
    await shot(p, '02-in-review-no-reviewer.png');

    const notices = p.locator('[data-testid=global-mutation-errors]');
    note('global notices on screen: ' + await notices.locator(':scope > *').count());
    for (const btn of await notices.getByRole('button').all()) { await btn.click().catch(() => undefined); }
    await p.locator('.ct-art button', { hasText: 'Name reviewers' }).click();
    await p.getByText(/Who should review/).waitFor({ timeout: 30000 });
    const offered = await p.locator('fieldset label.ct-art-row .ct-art-row-k').allInnerTexts();
    note(`offered as reviewers: ${JSON.stringify(offered)}`);
    await shot(p, '03-picker.png');
    await p.getByRole('checkbox', { name: /Reviewer User/ }).check();
    await p.getByLabel(/Note for the reviewers/).fill('Check the predicate device comparison.');
    await p.getByRole('button', { name: 'Assign' }).click();
    await p.getByText('Waiting for a decision').first().waitFor({ timeout: 30000 });
    note('reviewer named: Waiting for a decision');
    await shot(p, '04-reviewer-named.png');

    // ── 2. The named reviewer: where can they decide? ──
    const reviewer = await session('e2e-reviewer@trialsage.test');
    const pending = await (await fetch(`${BASE}/api/concept2cure/reviews/pending`, { headers: { Authorization: `Bearer ${reviewer.token}`, Origin: BASE } })).json();
    note(`reviewer GET /reviews/pending: ${JSON.stringify(pending).slice(0, 400)}`);
    const queue = await (await fetch(`${BASE}/api/concept2cure/reviews/my-queue`, { headers: { Authorization: `Bearer ${reviewer.token}`, Origin: BASE } })).json();
    note(`reviewer GET /reviews/my-queue: ${JSON.stringify(queue).slice(0, 400)}`);
    const r = await page(browser, reviewer);
    await r.p.goto(`${BASE}/concept2cure/review`, { waitUntil: 'domcontentloaded' });
    await r.p.getByText('Reviews you were named for').first().waitFor({ timeout: 120000 });
    note('reviewer Review screen: "Reviews you were named for" shown');
    await r.p.getByText(TITLE).first().click();
    await r.p.getByText(/SYNTHETIC WALK DATA, second pass/).first().waitFor({ timeout: 30000 });
    note('reviewer reads the version under review');
    note('reviewer sees Name reviewers: ' + await r.p.getByRole('button', { name: /Name reviewers/ }).count());
    await shot(r.p, '05-reviewer-request.png');
    await r.p.getByLabel(/You are a named reviewer/).fill('Predicate comparison is complete.');
    await r.p.getByRole('button', { name: 'Approve' }).click();
    await r.p.getByText(/Approved — “Predicate comparison is complete.”/).first().waitFor({ timeout: 30000 });
    note('reviewer decision recorded: Approved');
    await shot(r.p, '06-reviewer-approved.png');
    const after = await (await fetch(`${BASE}/api/concept2cure/projects/24/artifacts/${ARTIFACT}/reviews/status`, { headers: { Authorization: `Bearer ${admin.token}`, Origin: BASE } })).json();
    note('author reads the round: ' + JSON.stringify({ ready: after.data?.readyForApproval, reviewers: after.data?.reviewers?.map(x => [x.reviewerName, x.decision?.decision]) }));
  } catch (e) {
    note('WALK ERROR: ' + (e && e.stack ? e.stack.split('\n').slice(0, 3).join(' | ') : String(e)));
  } finally {
    fs.writeFileSync(path.join(OUT, 'walk.log'), log.join('\n') + '\n');
    await browser.close();
  }
})();
