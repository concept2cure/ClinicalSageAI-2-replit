import { mergeConfig } from 'vitest/config';
import { readFileSync } from 'node:fs';
import base from '/home/user/ClinicalSageAI-2-replit/vitest.config.ts';
const S = '/tmp/claude-0/-home-user-ClinicalSageAI-2-replit/787f5dd9-ee05-54b6-8bf9-81340ad671c0/scratchpad/d22';
const PROJECTS = '/home/user/ClinicalSageAI-2-replit/client/src/concept2cure/v2/surfaces/Projects.tsx';
const TEST = '/home/user/ClinicalSageAI-2-replit/client/src/concept2cure/v2/__tests__/newProjectWizardLook.test.tsx';
export default mergeConfig(base, {
  root: '/home/user/ClinicalSageAI-2-replit',
  plugins: [{
    name: 'serve-head', enforce: 'pre',
    load(id: string) {
      if (id.split('?')[0] === PROJECTS) return readFileSync(`${S}/Projects.HEAD.tsx`, 'utf8');
      return null;
    },
    transform(code: string, id: string) {
      if (id.split('?')[0] !== TEST) return null;
      const out = code.replace("resolve(__dirname, '../styles/project-home-v2.css')", `'${S}/project-home-v2.HEAD.css'`);
      if (out === code) throw new Error('css redirect did not apply');
      return out;
    },
  }],
});
