// Append to server/services/audit/__tests__/chain-anchor.dbtest.ts (it uses that
// file's memoryStore, writeChain, tamper, withClient and loadAnchor), then run
//   npx vitest run --config vitest.db.config.ts server/services/audit/__tests__/chain-anchor.dbtest.ts
describe('a newer anchor written after a truncation', () => {
  it('truncate the anchored head, then write the next anchor as the task role can: the truncation is still reported', async () => {
    const { writeAuditChainAnchor, verifyAuditChainAnchor } = await loadAnchor();
    const { store, objects } = memoryStore();
    const seven = await writeChain(7, 4);
    await withClient((c) => writeAuditChainAnchor(c, store, new Date('2031-01-01T02:00:00Z')));
    expect(await tamper('DELETE FROM audit_logs WHERE id = ANY($1::uuid[])', [seven.slice(2)])).toBe(2);
    expect((await withClient((c) => verifyAuditChainAnchor(c, store))).status).toBe('broken');
    // The task role holds PutObject on the anchors prefix: it writes tomorrow's anchor over the shorter chain.
    await withClient((c) => writeAuditChainAnchor(c, store, new Date('2031-01-02T02:00:00Z')));
    expect(objects.size).toBe(2);
    const verdict = await withClient((c) => verifyAuditChainAnchor(c, store));
    expect(verdict.status, JSON.stringify(verdict)).toBe('broken');   // at 735ba0a74: 'ok'
  });
});
