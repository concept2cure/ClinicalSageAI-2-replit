import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { stripTypeScriptTypes } from 'node:module';
import assert from 'node:assert/strict';
const raw = fs.readFileSync('/workspace/Concept2Cure/server/services/storage/local-provider.ts','utf8');
const source = raw.replace(/import \{[\s\S]*?\} from '\.\/storage-provider';/,'');
const js = stripTypeScriptTypes(source, {mode:'strip'});
const {LocalStorageProvider} = await import('data:text/javascript;base64,'+Buffer.from(js).toString('base64'));
const id = '11111111-1111-4111-8111-111111111111';
const dir = path.resolve('storage/vault/1/probe/versions',id);
fs.mkdirSync(dir,{recursive:true});
const original=Buffer.from('approved synthetic document');
const changed=Buffer.from('modified synthetic document');
const sha=b=>createHash('sha256').update(b).digest('hex');
fs.writeFileSync(path.join(dir,'synthetic.txt'),changed);
fs.writeFileSync(path.join(dir,'_meta.json'),JSON.stringify({
vaultFileId:'synthetic',vaultVersionId:id,filename:'synthetic.txt',mime:'text/plain',
sizeBytes:original.length,sha256:sha(original),metadata:{orgId:'1'},storedAt:new Date().toISOString()
}));
const provider=new LocalStorageProvider();
const got=await provider.get(id,1);
assert.ok(got);
assert.equal(got.bytes.toString(),changed.toString());
assert.equal(got.sha256,sha(original));
assert.notEqual(got.sha256,sha(got.bytes));
assert.equal(await provider.get(id,2),null);
