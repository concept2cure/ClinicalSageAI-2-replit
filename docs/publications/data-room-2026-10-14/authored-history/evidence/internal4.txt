### 2. Apply the schema (idempotent)

> **Run this whole block in one shot.** It is safe to re‑run (uses `if not exists` & guarded `alter table`).

```sql
-- ===============================
-- CMC Document Authoring Core DDL
-- Safe to re-run (idempotent)
-- ===============================

-- 0) housekeeping helpers
create or replace function set_updated_at()
returns trigger language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end$$;

-- -------------------------------
-- 1) Documents and Sections
-- -------------------------------
create table if not exists doc_documents (
  doc_id       uuid primary key default gen_random_uuid(),
  title        text not null,
  module       text not null,                -- e.g., 'M3'
  product_code text,
  status       text not null default 'DRAFT',-- DRAFT/IN_REVIEW/APPROVED/ARCHIVED
  locale       text not null default 'en',
  created_by   text,
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now(),
  check (status in ('DRAFT','IN_REVIEW','APPROVED','ARCHIVED'))
);

create table if not exists doc_sections (
  section_id   uuid primary key default gen_random_uuid(),
  doc_id       uuid not null references doc_documents(doc_id) on delete cascade,
  code         text not null,                -- e.g., '3.2.S.1'
  title        text not null,
  order_idx    int not null default 0,
  content      jsonb not null default '{}'::jsonb,  -- editor JSON (TipTap/ProseMirror)
  track_changes boolean not null default false,
  updated_by   text,
  updated_at   timestamptz not null default now(),
  unique (doc_id, code)
);

do $$
begin
  if not exists (select 1 from pg_trigger where tgname='trg_doc_documents_updated_at') then
    create trigger trg_doc_documents_updated_at
      before update on doc_documents
      for each row execute function set_updated_at();
  end if;

  if not exists (select 1 from pg_trigger where tgname='trg_doc_sections_updated_at') then
    create trigger trg_doc_sections_updated_at
      before update on doc_sections
      for each row execute function set_updated_at();
  end if;
end$$;

create index if not exists idx_doc_sections_doc on doc_sections(doc_id);

-- -------------------------------
-- 2) Revisions (append-only) & Comments
-- -------------------------------
create table if not exists doc_revisions (
  rev_id     uuid primary key default gen_random_uuid(),
  section_id uuid not null references doc_sections(section_id) on delete cascade,
  snapshot   jsonb not null,                 -- full content snapshot
  diff       jsonb,                          -- optional minimal patch
  author     text,
  comment    text,
  created_at timestamptz not null default now()
);

-- forbid updates/deletes to keep audit immutable
do $$
begin
  if not exists (select 1 from pg_trigger where tgname='trg_doc_revisions_block_ud') then
    create trigger trg_doc_revisions_block_ud
      before update or delete on doc_revisions
      for each row execute function pg_catalog.raise_exception('doc_revisions is append-only');
  end if;
exception when duplicate_object then
  -- ignore if it already exists
  null;
end$$;

create table if not exists doc_comments (
  comment_id  uuid primary key default gen_random_uuid(),
  section_id  uuid not null references doc_sections(section_id) on delete cascade,
  author      text,
  body        text not null,
  anchor      jsonb,                          -- editor position (from,to,attr)
  status      text not null default 'OPEN',   -- OPEN/RESOLVED
  created_at  timestamptz not null default now(),
  resolved_by text,
  resolved_at timestamptz,
  check (status in ('OPEN','RESOLVED'))
);

-- -------------------------------
-- 3) Citations (token snapshots) & Guidance cache
-- -------------------------------
create table if not exists doc_citations (
  cite_id     uuid primary key default gen_random_uuid(),
  section_id  uuid not null references doc_sections(section_id) on delete cascade,
  anchor      jsonb,                          -- where citation sits in content
  source      text not null,                  -- e.g., 'QUALITY.SPECS.DP'
  payload     jsonb not null,                 -- frozen snapshot (for traceability)
  created_at  timestamptz not null default now()
);

create table if not exists doc_guidance_cache (
  guide_id     uuid primary key default gen_random_uuid(),
  section_code text not null,                 -- e.g., '3.2.S.1'
  region       text not null default 'ICH',   -- ICH/FDA/EMA/PMDA
  guidance_md  text not null,                 -- rendered markdown
  deficiencies jsonb not null default '[]'::jsonb,
  updated_at   timestamptz not null default now(),
  unique (section_code, region)
);

create index if not exists idx_doc_citations_section on doc_citations(section_id);

-- -------------------------------
-- 4) Safe alignment (only if columns missing)
-- -------------------------------
-- examples; these "alter if missing" blocks are harmless if already present
do $$
begin
  if not exists (select 1 from information_schema.columns where table_name='doc_sections' and column_name='track_changes') then
    alter table doc_sections add column track_changes boolean not null default false;
  end if;
  if not exists (select 1 from information_schema.columns where table_name='doc_documents' and column_name='locale') then
    alter table doc_documents add column locale text not null default 'en';
  end if;
end$$;

-- ===============================
-- END OF DDL
-- ===============================
```
